def predicted_metrics(q):
    f=np.array(q['frequency_GHz']);s11=np.array(q['S11_dB']);s21=np.array(q['S21_dB'])
    transmission=intervals(f,np.full(len(f),-1000.),s21)
    band=max(transmission,key=lambda b:b[1]-b[0]);ii=find_peaks(-s11,prominence=.5)[0];ii=[int(i) for i in ii if band[0]<f[i]<band[1] and s11[i]<-10]
    result=dict(S21_BW_GHz=band[1]-band[0],S21_band_GHz=band,central_joint20_band_GHz=center_band(intervals(f,s11,s21)),prominent_valley_count=len(ii),derived_only_from_frozen_curve=True)
    if len(ii)==3:
        peaks=[int(a+1+np.argmax(s11[a+1:b])) for a,b in zip(ii,ii[1:])]
        result.update(central_valley_GHz=float(f[ii[1]]),inner_peaks=[dict(GHz=float(f[i]),dB=float(s11[i])) for i in peaks],worse_inner_peak_dB=float(max(s11[peaks])),peak_difference_dB=float(abs(np.diff(s11[peaks])[0])))
    return result

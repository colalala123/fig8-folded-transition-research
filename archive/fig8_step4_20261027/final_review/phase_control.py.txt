import hashlib,json,os
from pathlib import Path
from datetime import datetime,timezone
HERE=Path(__file__).resolve().parent
PARENT=HERE.parent
ORIGINAL=PARENT.parent/'fig8_finish24_20261023'
BASE=Path('J:/f8_B6_261027')

ORDER=['B01', 'B02', 'B03', 'B04', 'B05', 'B06']
def read(path):return json.loads(Path(path).read_text(encoding='utf-8-sig'))
def sha(path):return hashlib.sha256(Path(path).read_bytes()).hexdigest()
def now():return datetime.now(timezone.utc).isoformat()
def write(path,data):
    path=Path(path);path.parent.mkdir(parents=True,exist_ok=True)
    tmp=path.with_suffix(path.suffix+'.tmp');tmp.write_text(json.dumps(data,ensure_ascii=False,indent=2,default=str,allow_nan=False),encoding='utf-8');os.replace(tmp,path)
def status_report():
    active=read(HERE/'active.json')
    write(HERE/'status.json',dict(at=now(),phase='B6D',active=active,cases=ORDER))
    (HERE/'STATUS.md').write_text('# 18配置同网格直接复核\n\n'+json.dumps(active,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
    # Runtime files are intentionally not part of the frozen scientific inputs.
    # This lets the initial completion watch recognize the real successor solve.
    write(PARENT/'active.json',dict(**active,phase='B6D'))
    write(PARENT/'status.json',dict(at=now(),stage='B6D',active=active,cases=ORDER))
    (PARENT/'STATUS.md').write_text('# Fig8 18配置：最终同网格直接复核\n\n'+json.dumps(active,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')

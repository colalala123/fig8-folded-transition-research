import copy
import tempfile
import unittest
from pathlib import Path
from phase_direct_context import budget_guard,sha

class DirectBudgetTests(unittest.TestCase):
    def setUp(self):
        t=tempfile.TemporaryDirectory();self.addCleanup(t.cleanup)
        p=Path(t.name)/'input.json';p.write_text('{}')
        self.p=dict(case='D09G05',role='cached_phase_direct',cached_task_ordinal_for_case=1,geometry_budget_increment=0,
            adaptive_allowed=False,input_path=str(p),historical_geometry_attempts=7,new_geometry_used=3,lifetime_geometry_attempts=10)
        self.a=dict(execution_authorized=True,stage='B',stage_C_released=False,geometry_cap=6,cached_task_cap=2)
        tasks=[dict(case='D09G05',role=r,new_phase_geometry_ordinal=i+1,status='reviewed') for i,r in enumerate(['probe_Ls1','probe_Lm','candidate'])]
        tasks.append(dict(case='D09G05',role='cached_phase_direct',input_sha256=sha(p),path=str(p),status='reserved'))
        self.l=dict(case_cumulative_geometry={'D04G02':6,'D09G05':7},new_geometry_reserved=3,new_cached_tasks_reserved=1,tasks=tasks)
    def test_one_cache_allowed(self):budget_guard(self.p,self.l,self.a)
    def test_adaptive_forbidden(self):
        self.p['adaptive_allowed']=True
        with self.assertRaises(AssertionError):budget_guard(self.p,self.l,self.a)
    def test_geometry_increment_forbidden(self):
        self.p['geometry_budget_increment']=1
        with self.assertRaises(AssertionError):budget_guard(self.p,self.l,self.a)
    def test_history_not_zeroed(self):
        self.p['historical_geometry_attempts']=0
        with self.assertRaises(AssertionError):budget_guard(self.p,self.l,self.a)
    def test_unreviewed_source_forbidden(self):
        self.l['tasks'][2]['status']='reserved'
        with self.assertRaises(AssertionError):budget_guard(self.p,self.l,self.a)
    def test_second_cache_same_case_forbidden(self):
        t=copy.deepcopy(self.l['tasks'][-1]);t['path']+='other'
        self.l['tasks'].append(t);self.l['new_cached_tasks_reserved']=2
        with self.assertRaises(AssertionError):budget_guard(self.p,self.l,self.a)
    def test_C_forbidden(self):
        self.p['case']='D12G05'
        with self.assertRaises(AssertionError):budget_guard(self.p,self.l,self.a)
    def test_lifetime_not_reset(self):
        self.p['lifetime_geometry_attempts']=3
        with self.assertRaises(AssertionError):budget_guard(self.p,self.l,self.a)

if __name__=='__main__':unittest.main()

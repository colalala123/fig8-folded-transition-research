"""Stage B accounting. Original and stage A budgets remain frozen."""
from pathlib import Path
import sys
from phase_control import HERE,BASE,ORIGINAL,PLAN,read,write,sha,now,status
sys.path.insert(0,str(ORIGINAL))
OLD=Path('J:/f8_261007')
status_report=status
HISTORY={'D04G02':6,'D09G05':7}
STEPS=['probe_Ls1','probe_Lm','candidate']

def validate_budget(spec,ledger,authorization):
    case=spec['case'];n=spec['new_phase_geometry_ordinal']
    assert authorization['execution_authorized'] and authorization['stage']=='B' and not authorization['stage_C_released']
    assert case in HISTORY and n in (1,2,3) and spec['role']==STEPS[n-1]
    assert spec['historical_geometry_attempts']==HISTORY[case]
    assert spec['phase_solve_ordinal_including_history']==HISTORY[case]+n
    assert not spec.get('resume_checkpoint'),'Recovery needs a separate evidenced decision'
    assert ledger['case_cumulative_geometry']==HISTORY
    tasks=[t for t in ledger['tasks'] if t.get('new_phase_geometry_ordinal')]
    assert len(tasks)==ledger['new_geometry_reserved']<=authorization['geometry_cap']==6
    assert 0<=ledger['new_cached_tasks_reserved']<=authorization['cached_task_cap']==2
    assert len({t['path'] for t in tasks})==len(tasks)
    assert all(t['case'] in HISTORY for t in tasks)
    for c in HISTORY:
        ct=[t for t in tasks if t['case']==c]
        assert sorted(t['new_phase_geometry_ordinal'] for t in ct)==list(range(1,len(ct)+1)) and len(ct)<=3
    own=[t for t in tasks if Path(t['path']).resolve()==Path(spec['input_path']).resolve()]
    assert len(own)==1 and own[0]['case']==case and own[0]['new_phase_geometry_ordinal']==n
    assert own[0]['input_sha256']==sha(spec['input_path'])

def validate_geometry_budget(spec,spec_path):
    assert not (HERE/'stage_B_closure.json').exists()
    assert Path(spec_path).resolve()==Path(spec['input_path']).resolve()
    expected=BASE/spec['case']/f"g0{spec['new_phase_geometry_ordinal']}_{spec['role']}"
    assert Path(spec['run_dir']).resolve()==expected.resolve()
    assert sha(HERE/'phase_geometry_worker.py')==spec['worker_sha256']
    assert sha(ORIGINAL/'mesh_budget.py')==spec['controller_sha256']
    validate_budget(spec,read(HERE/'budget_ledger.json'),read(HERE/'authorization.json'))

def read(p): return json.loads(Path(p).read_text(encoding='utf-8-sig'))
def sha(p): return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def write(p, d):
    p = Path(p); p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(json.dumps(d, ensure_ascii=False, indent=2, allow_nan=False), encoding='utf-8')
def now(): return datetime.now(timezone.utc).isoformat()
def resource14(s):
    return (s.replace("'cores_per_task':16", "'cores_per_task':14")
            .replace('-max_processors 16', '-max_processors 14')
            .replace('16core', '14core').replace('16-core', '14-core'))

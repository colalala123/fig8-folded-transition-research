from pathlib import Path
import psutil
import phase_control as pc
def verify():
    identity=pc.read(pc.HERE/'aedt_identity.json');p=psutil.Process(identity['pid'])
    assert p.is_running() and abs(p.create_time()-identity['created'])<.01 and p.name().lower()=='ansysedt.exe' and p.cmdline()==identity['command']
    desktops=[x for x in psutil.process_iter(['name']) if (x.info['name'] or '').lower()=='ansysedt.exe'];assert len(desktops)==1
    return identity
def install(dc):
    original=dc.app_for
    def guarded(path):
        verify();assert Path(path).resolve().is_relative_to(pc.BASE.resolve());return original(path)
    dc.app_for=guarded

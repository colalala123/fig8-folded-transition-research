inputs={

 'configuration':ROOT/'output/fig8_iso4_20261026/cases/I01G02/configuration.json',
 'microstrip_samples':ROOT/'txline_microstrip_samples.json',
 'slotline_samples':ROOT/'txline_slotline_samples.json',

 'model_implementation':OUT/'line_models.py',
}

cad=read(inputs['configuration'])
assert cad['passed'] and cad['material']['er']==3.55 and cad['material']['h_mm']==.813
thicknesses=[abs(b[5]-b[2]) for b in cad['microstrip_boxes'].values()]
assert all(abs(t-.035)<1e-8 for t in thicknesses)
base=[{k:r[k] for k in ['case','group','Wm','Ws','a','BW_GHz','mesh','near_100k_native','peak_level_dB']} for r in old['rows']]
assert len(base)==34 and len(set(r['group'] for r in base))==17

# Reproduce old raw TXLINE readings within the present geometry range.
# Compare directly, not to a fitted polynomial and not to HFSS bandwidth.
regression=[]
for key,width_key,func,lo,hi in [('microstrip_samples','width_um',microstrip,.9,2.),('slotline_samples','gap_um',slotline,.30,.58)]:
 for r in read(inputs[key]):
  w=r[width_key]/1000
  if not lo<=w<=hi:continue
  z=func(3.38,w,f=6.85)['Z_ohm']
  regression.append(dict(kind=key,width_mm=w,TXLINE_old_ohm=r['impedance_ohm'],analytical_ohm=z,
   error_ohm=z-r['impedance_ohm'],error_percent=100*(z/r['impedance_ohm']-1)))
assert len(regression)==6
assert max(abs(r['error_percent']) for r in regression)<.1

def map_rows(er=3.55,f=6.72,dispersion=True,t=.035):
 ans=[]
 for r in base:
  m=microstrip(er,r['Wm'],t=t,f=f,dispersion=dispersion);s=slotline(er,r['Ws'],f=f)
  zm=m['Z_ohm'];zs=s['Z_ohm']
  ans.append(dict(r,Zm_nominal_ohm=zm,Zs_nominal_ohm=zs,P_nominal=zm*zs/2500,R_nominal=zm/zs))
 return ans

def paired(rows,tol):
 ans=[]
 for x,y in itertools.combinations(rows,2):
  if x['a']!=y['a']:continue
  dp=abs(x['P_nominal']-y['P_nominal'])/((x['P_nominal']+y['P_nominal'])/2)
  lo,hi=sorted([x,y],key=lambda r:r['R_nominal'])
  ratio=hi['R_nominal']/lo['R_nominal']
  if dp>tol or ratio<1.1:continue
  ans.append(dict(low_R_case=lo['case'],high_R_case=hi['case'],a=lo['a'],
   P_mismatch_percent=100*dp,R_ratio=ratio,BW_highR_minus_lowR_MHz=1000*(hi['BW_GHz']-lo['BW_GHz']),
   supports_decrease=hi['BW_GHz']<lo['BW_GHz'],both_clean=lo['near_100k_native'] and hi['near_100k_native']))
 return sorted(ans,key=lambda p:(p['P_mismatch_percent'],-p['R_ratio']))

def design(rows,only=False):
 return np.array([[1,math.log(r['P_nominal'])]+([] if only else [math.log(r['R_nominal'])])+[r['a']-1.6] for r in rows])
def fit(rows,only=False):
 return np.linalg.lstsq(design(rows,only),[r['BW_GHz'] for r in rows],rcond=None)[0]
def errors(true,pred):
 e=(np.array(pred)-np.array(true))*1000
 return dict(MAE_MHz=float(np.mean(abs(e))),RMSE_MHz=float(np.sqrt(np.mean(e*e))),max_abs_MHz=float(max(abs(e))))
def stats(rows):
 train=rows[:24];test=rows[24:];models={}
 for key,only in [('P_only',True),('P_and_R',False)]:
  b=fit(train,only);truth=[];prediction=[];slopes=[]
  for group in sorted(set(r['group'] for r in train)):
   tr=[r for r in train if r['group']!=group];te=[r for r in train if r['group']==group]
   bb=fit(tr,only);truth.extend(r['BW_GHz'] for r in te);prediction.extend(design(te,only)@bb)
   if not only:slopes.append(float(bb[2]))
  models[key]=dict(coefficients=b.tolist(),leave_group_out=errors(truth,prediction),
   heldout10=errors([r['BW_GHz'] for r in test],design(test,only)@b),leave_group_R_slopes=slopes)
 return dict(models=models,clean16_coefficients=fit([r for r in train if r['near_100k_native']]).tolist(),
  matching={str(t):paired(rows,t) for t in [.005,.01,.02,.05]})

rows=map_rows();results=stats(rows)
scenarios={}
for name,kw in {
 'same_frequency_old_material_6p72':dict(er=3.38,f=6.72),
 'same_analytical_models_old_material_6p85':dict(er=3.38,f=6.85),
 'current_material_6p85':dict(f=6.85),
 'current_material_6p69':dict(f=6.69),
 'current_material_6p75':dict(f=6.75),
 'current_material_microstrip_quasistatic':dict(dispersion=False),
 'current_material_zero_microstrip_thickness_sensitivity_only':dict(t=0),
}.items():
 rr=map_rows(**kw);ss=stats(rr)
 scenarios[name]=dict(settings=kw,R_coefficient=ss['models']['P_and_R']['coefficients'][2],
  heldout10=ss['models']['P_and_R']['heldout10'],
  matching={k:dict(n=len(p),decreasing=sum(x['supports_decrease'] for x in p),pairs=p) for k,p in ss['matching'].items()})

checks=[]
for func,w in [(microstrip,1.2),(slotline,.44)]:

 assert abs(a-b)<1e-9
 assert abs(a-func(3.38,w)['Z_ohm'])>.01
 checks.append(func.__name__+': scale invariance and material sensitivity passed')
for er,w,h,f in [(4,.44,.813,6.72),(3.55,.0001,.813,6.72),(3.55,.44,.0001,6.72)]:
 try:slotline(er,w,h,f)
 except ValueError:pass
 else:raise AssertionError('Slotline range gate failed')

 settings=dict(er=3.55,h_mm=.813,f_reference_GHz=6.72,microstrip_t_mm=.035,slotline_t_model_mm=0,
  microstrip_model='Hammerstad/Jensen finite-thickness + Kirschning/Jansen dispersion, Qucs width convention',
  slotline_model='Janaswamy/Schaubert eq 9, zero-metal-thickness standard slotline',
  tan_delta_in_HFSS=.0027,impedance_convention='real nominal lossless line-model impedance; no lossy-complex or folded-effective extraction'),
 measured_microstrip_thickness_mm=thicknesses,old_TXLINE_regression=regression,rows=rows,**results,
 impedance_tables=dict(
  microstrip=[dict(width_mm=w,**microstrip(3.55,w)) for w in sorted(set(r['Wm'] for r in rows))],
  slotline=[dict(width_mm=w,**slotline(3.55,w)) for w in sorted(set(r['Ws'] for r in rows))]),
 sensitivity_scenarios=scenarios,unit_checks=checks,
 source_paths={k:str(p) for k,p in inputs.items()},source_hashes={k:hashlib.sha256(p.read_bytes()).hexdigest() for k,p in inputs.items()},
 bibliography=[
 'https://ntrs.nasa.gov/api/citations/19860021419/downloads/19860021419.pdf',
 'https://qucs.sourceforge.net/tech/node75.html',
 'https://qucs.sourceforge.net/doxygen/0.0.18/qucs-core/msline_8cpp_source.html',
 'https://raw.githubusercontent.com/scikit-rf/scikit-rf/master/skrf/media/mline.py'],
 no_HFSS_simulation_started=True,no_response_inversion=True,new_TXLINE_GUI_readings=False,
 caveats=['Nominal standard-line impedances are not the effective folded/sandwiched coupled network impedances.',
  'Old TXLINE agreement checks implementation consistency, not accuracy against real current-material structures.',

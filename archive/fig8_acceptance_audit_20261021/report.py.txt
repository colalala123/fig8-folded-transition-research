def decision(a,direct=None):
    if a.get('offline_category')=='sweep_not_converged':return 'needs_sweep_repair'
    conv=a.get('convergence') or {}
    if not (a.get('complete_finite_grid') and a.get('native_curve_hash_matches_manifest') and a.get('source_hash_matches_manifest') and a.get('port_passed') and conv.get('converged') and conv.get('final_delta_s',1)<=.005):
        return 'needs_source_port_or_convergence_review'
    if a.get('offline_category')=='full_matrix_pass':return 'existing_numerical_pass'
    if direct and direct.get('sample_response_passed') and direct.get('band_edges_passed'):
        return 'fundamental_response_checked_needs_geometry_tuning'
    if direct:return 'fundamental_discrepancy_needs_targeted_repair'
    return 'candidate_needs_case_specific_direct_check'

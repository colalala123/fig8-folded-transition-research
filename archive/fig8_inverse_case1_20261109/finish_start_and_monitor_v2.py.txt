import argparse, json, subprocess, sys, time
from pathlib import Path
import psutil
from build_runtime import ROOT, read, write, new, sha, now

def exact(record):
    try:
        p = psutil.Process(record['pid'])
        return p.is_running() and abs(p.create_time() - record['created']) < 0.01 and (p.cmdline() == record['command'])
    except psutil.NoSuchProcess:
        return False

def ensure_observer(ctx, run):
    sys.path.insert(0, str(ctx))
    from phase_heartbeat import mutex
    path = run / 'lifecycle_observer/daemon.json'
    folder = path.parent
    folder.mkdir(exist_ok=True)
    with mutex(folder):
        if path.exists():
            previous = read(path)
            if exact(previous):
                return previous
        args = [sys.executable, str(ROOT / 'observer_retry_v2.py'), '--context', str(ctx), '--run', str(run)]
        with (folder / 'daemon_retry_v2.log').open('ab') as log:
            p = subprocess.Popen(args, cwd=str(ctx), stdin=subprocess.DEVNULL, stdout=log, stderr=log, creationflags=subprocess.CREATE_NO_WINDOW)
        q = psutil.Process(p.pid)
        record = dict(pid=q.pid, created=q.create_time(), command=q.cmdline(), registered_at=time.time())
        write(path, record)
    for i in range(40):
        if path.exists():
            record = read(path)
            if exact(record) and 'observer_retry_v2.py' in ' '.join(record['command']):
                return record
        time.sleep(0.25)
    raise RuntimeError('Observer-only adapter did not become live; do not relaunch solver')

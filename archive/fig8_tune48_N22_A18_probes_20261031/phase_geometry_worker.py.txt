async def solve_stage(project, run, state, label, adaptive):
    """Run only the named setup, with a bounded wall clock; never kill/retry on timeout."""
    assert not dc.solvers(),'Another HFSS solver active'
    assert len(state['segments'])<13, 'bounded geometry Analyze cap exhausted'
    assert sum(bool(x['adaptive_requested']) for x in state['segments'])<12 if adaptive else sum(not x['adaptive_requested'] for x in state['segments'])<1
    validate_campaign(extra_analyze=1,case=state['case'])
    args={'project_name':project.stem,'design_name':'Wideband vertical transition','setup_name':'Setup19','port':7990,
          'concurrent_tasks':1,'cores_per_task':14,'blocking':False,'timeout':120}
    item={'label':label,'adaptive_requested':adaptive,'started_at':now(),'mcp_request':args,'observations':[]}
    state['segments'].append(item);write(run/'state.json',state)
    start=time.monotonic()
    async with dc.connected_client() as client:
        item['mcp_response']=await dc.session_call(client,'start_analysis',args)
        assert item['mcp_response'].get('started')
        state['status']='solving';write(run/'state.json',state)
        while True:
            await asyncio.sleep(5);live=dc.solvers()
            if live and (not item['observations'] or item['observations'][-1]!=live):item['observations'].append(live)
            if not adaptive and any('-solve_broadband_adapt' in ' '.join(p['cmdline'] or []) or '-solve_adapt ' in ' '.join(p['cmdline'] or []) for p in live):
                app=dc.app_for(project)
                try:app.stop_simulations(clean_stop=True)
                finally:app.release_desktop(close_projects=False,close_desktop=False)
                raise RuntimeError('Unexpected new adaptive solve during final sweep; clean stop requested')
            if live:
                # Do not flood the AEDT broker while hf3d is visibly working.
                item['last_status']={'running':True,'source':'observed_hf3d_process'}
            else:
                try:
                    item['last_status']=await dc.session_call(client,'get_analysis_status',{k:v for k,v in args.items() if k in ['project_name','design_name','setup_name','port','timeout']})
                except Exception as exc:
                    # A read-only status failure is NOT evidence that Analyze failed.
                    # Never re-send start_analysis. One direct read fallback per stage.
                    if item.get('direct_status_fallback'):raise
                    item['direct_status_fallback']={'at':now(),'broker_error':str(exc)}
                    write(run/'state.json',state)
                    app=dc.app_for(project)
                    try:
                        running=bool(app.are_there_simulations_running)
                        item['last_status']={'running':running,'source':'direct_aedt_after_broker_timeout'}
                        item['direct_status_fallback']['messages']=list(app.odesktop.GetMessages(app.project_name,app.design_name,0))
                    finally:app.release_desktop(close_projects=False,close_desktop=False)
            item['elapsed_seconds']=time.monotonic()-start;write(run/'state.json',state)
            if not live and not item['last_status'].get('running'):break
            if time.monotonic()-start>3600:raise TimeoutError('Stage supervision limit; preserve live solver and inspect, no automatic retry')
    item['finished_at']=now()
    commands=[' '.join(p['cmdline'] or []) for obs in item['observations'] for p in obs]
    item['observed_14core_process']=any('-max_processors 14' in c and '-numParallelGroups 1' in c for c in commands)
    item['observed_adaptive_process']=any('-solve_broadband_adapt' in c or '-solve_adapt ' in c for c in commands)
    if adaptive:assert item['observed_14core_process'] and item['observed_adaptive_process']
    write(run/'state.json',state)
    return item

async def main(spec_path):
    spec=read(spec_path);case=spec['case'];run=Path(spec['run_dir']);job=HERE/'cases'/case
    validate_geometry_budget(spec,spec_path)  # Before any locks, state changes or failure hook.
    assert not dc.solvers() and not (OLD/'pilot_queue.lock').exists()
    locks=[HERE/'worker.lock',OLD/'protocol.lock'];assert all(not p.exists() for p in locks)
    acquired=[]
    try:
        for lock in locks:
            with lock.open('x') as h:h.write(str(os.getpid()))
            acquired.append(lock)
    except BaseException:
        for p in acquired:p.unlink()
        raise
    run.mkdir(parents=True,exist_ok=True);job.mkdir(parents=True,exist_ok=True);status='failed'
    try:
        write(HERE/'active.json',{'case':case,'stage':'supervised_geometry_prepare','pid':os.getpid(),'run_dir':str(run),'started_at':now()});status_report()
        result=await execute(spec_path);status='completed';metrics=result['metrics']
        summary=f"{case}新几何求解完成：{result['actual_elements']}单元；原5.6GHz联合目标={'通过、待复核' if result['design_passed'] else '未通过，按有限校正方案评估'}；BW={metrics.get('passband_2dB',{}).get('BW_GHz')}GHz。不是完整验收。"
        artifacts=['finish24_result.json','state.json','manifest.json','response_metrics.json','sparams.csv','all_modes_sparams.csv','mode_impedance_gamma.csv','convergence.txt','input.json']
        artifacts+=['reference_review.json','target20_metrics.json']
    except BaseException:
        write(run/'worker_error.json',{'at':now(),'error':traceback.format_exc(),'live_solvers':dc.solvers()})
        write(job/'attention.json',{'at':now(),'error_file':str(run/'worker_error.json')})
        summary=f'{case}新几何工作项需检查worker_error.json；保留现有网格/结果，不自动重复求解。';artifacts=['worker_error.json','state.json'] if (run/'state.json').exists() else ['worker_error.json']
    finally:
        for lock in acquired:
            if lock.exists() and lock.read_text()==str(os.getpid()):lock.unlink()
        write(HERE/'active.json',{'case':case,'stage':'awaiting_geometry_hook_review','operation_status':status,'run_dir':str(run),'finished_at':now()});status_report()
    try:write(run/'notification_receipt.json',await publish(run,case,status,summary,artifacts))
    except Exception:write(run/'notification_error.json',{'at':now(),'error':traceback.format_exc()});raise

if __name__=='__main__':
    parser=argparse.ArgumentParser();parser.add_argument('--spec',type=Path,required=True);args=parser.parse_args();asyncio.run(main(args.spec.resolve()))

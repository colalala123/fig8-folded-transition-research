"""Resume the original supervised queue, with 600-second worker monitoring only."""
import argparse
import json
import os
import subprocess
import sys
from pathlib import Path
import psutil
import guarded_pilot_queue as queue
import continue_d04_after_i03 as continuation

HERE = Path(__file__).resolve().parent
RECORD = queue.ROOT / 'supervised_queue_600.json'


class WorkerTransport:
    STDOUT = subprocess.STDOUT

    def call(self, args, **kwargs):
        assert Path(args[1]).resolve() == HERE / 'run_guarded_case.py'
        assert args[-1] == '--solve'
        return subprocess.call([args[0], str(HERE / 'run_guarded_case_600.py'), *args[2:]], **kwargs)


def readiness():
    ready = continuation.readiness()
    assert ready['ready']
    from resume_i03_after_timeout import validate_resumption
    validate_resumption()
    old = queue.read(queue.ROOT / 'pilot_queue_state.json')
    assert old['status'] == 'review_required' and old['active_case'] == 'D04G05'
    assert Path(old['active_run']) == queue.ROOT / 'D04G05/i03'
    assert not psutil.pid_exists(old['pid'])
    continuation.base.idle()
    assert not RECORD.exists(), 'Existing queue adapter requires review'
    metrics = queue.read(queue.ROOT / 'D04G05/i03/verified_recovery/response_metrics.json')
    protocol = queue.read(queue.ROOT / 'production_protocol.json')
    assert queue.q.accepted(metrics, protocol['central_valley_target_GHz'])
    return ready, old


def main():
    ready, old = readiness()
    files = [Path(__file__).resolve(), HERE / 'run_guarded_case_600.py']
    record = {'status': 'running', 'pid': os.getpid(), 'interval_seconds': 600,
        'started_at_utc': queue.now(), 'prior_queue_state': old, 'readiness': ready,
        'stage': 'supervised', 'remaining_released': False,
        'implementation_hashes': {str(p): queue.sha(p) for p in files}}
    with RECORD.open('x', encoding='utf-8') as stream:
        json.dump(record, stream, indent=2)
    queue.subprocess = WorkerTransport()
    sys.argv = [str(HERE / 'guarded_pilot_queue.py'), '--stage', 'supervised', '--solve']
    try:
        queue.main()
        record['status'] = 'supervised_review_required'
    except BaseException as exc:
        record.update(status='review_required', error=repr(exc))
        raise
    finally:
        record['finished_at_utc'] = queue.now()
        queue.write(RECORD, record)


if __name__ == '__main__':
    ap = argparse.ArgumentParser()
    ap.add_argument('--run', action='store_true')
    args = ap.parse_args()
    if args.run:
        main()
    else:
        readiness()
        print(json.dumps({'ready': True, 'stage': 'supervised', 'poll_seconds': 600}))

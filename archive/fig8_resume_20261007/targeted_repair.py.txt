import json, hashlib, subprocess, sys
from pathlib import Path

HERE = Path(__file__).resolve().parent
ROOT = Path('J:/f8_261007')
PLAN = json.loads((HERE/'resume_queue_plan.json').read_text(encoding='utf-8'))
PROTOCOL = json.loads((ROOT/'production_protocol.json').read_text(encoding='utf-8'))
SOURCE = Path('F:/hfss_f8_261002/B0/B0.aedt')

targets = ['D03G02']
for case in PLAN['cases']:
    if case['case'] not in targets:
        continue
    folder = ROOT/case['case']
    existing = sorted(folder.glob('i*/state.json'))
    ordinal = case['historical_solves'] + len(existing)
    if ordinal >= 8:
        raise RuntimeError(f'{case["case"]}: no phase budget remains')
    run = folder/f'i{ordinal:02d}'
    run.mkdir(parents=True, exist_ok=True)
    spec = {
        'case': case['case'], 'geometry': case['selected_geometry'],
        'role': 'targeted_numerical_repair', 'source': str(SOURCE),
        'source_sha256': hashlib.sha256(SOURCE.read_bytes()).hexdigest(),
        'protocol_id': PROTOCOL['protocol_id'],
        'protocol_sha256': hashlib.sha256((ROOT/'production_protocol.json').read_bytes()).hexdigest(),
        'cores': 16, 'tasks': 1, 'prediction': None,
        'parent_curve_sha256': None, 'frozen_at_utc': '2026-10-09T00:00:00Z'
    }
    frozen = run/'prediction_frozen.json'
    if frozen.exists():
        old = json.loads(frozen.read_text(encoding='utf-8'))
        if old['geometry'] != spec['geometry']:
            raise RuntimeError(f'{case["case"]}: frozen geometry mismatch')
    else:
        frozen.write_text(json.dumps(spec, indent=2), encoding='utf-8')
    log = run/'targeted_repair_worker.log'
    print(f'RUN {case["case"]} {run}', flush=True)
    py = Path('${WORKSPACE}/.venv/Scripts/python.exe')
    rc = subprocess.call([str(py), str(HERE/'run_guarded_case.py'), str(frozen), '--solve'], cwd=str(HERE.parent), stdout=log.open('a'), stderr=subprocess.STDOUT)
    if rc:
        raise SystemExit(f'{case["case"]} worker exit {rc}; inspect {log}')
    print(f'COMPLETE {case["case"]}', flush=True)

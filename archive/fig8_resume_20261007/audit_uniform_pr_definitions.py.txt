"""Read-only provenance and independent formula check; no simulations."""
import csv,json,sys
from pathlib import Path
from datetime import datetime,timezone
HERE=Path(__file__).resolve().parent
WORK=HERE.parent.parent
sys.path[:0]=[str(WORK),str(WORK/'output/folded_simulation_plan_20260920')]
from fit_modified_impedances import feature_based_impedances,read_csv_rows,ensure_db_columns
from response_metrics import analyze
from guarded_pilot_queue import sha


def main():
    selection=Path('F:/2026-1/circuit/mcp/opt/folded_r8_20260814/selection.json')
    chosen=json.loads(selection.read_text(encoding='utf-8'))
    index=Path(chosen['source_index'])
    with index.open(encoding='utf-8-sig',newline='') as h:uniform=list(csv.DictReader(h))
    assert len(uniform)==1000
    seeds=[]
    for case in chosen['cases']:
        matches=[row for row in uniform if row['run']==case['run']]
        assert len(matches)==1
        row=matches[0];zm,zs=float(row['Zm_prime_ohm']),float(row['Zs_prime_ohm'])
        p,r=zm*zs/2500,zm/zs
        assert abs(p-float(row['P_eff_from_Zm_prime_Zs_prime']))<5.1e-7
        assert abs(r-float(row['R_eff_from_Zm_prime_Zs_prime']))<5.1e-7
        assert abs(p-case['uniform_P_eff'])<5.1e-7
        assert abs(r-case['uniform_R_eff'])<5.1e-7
        seeds.append({'folded_seed':case['id'],'uniform_run':case['run'],
            'P_from_index_impedances':p,'R_from_index_impedances':r,
            'label_match':True,'original_uniform_curve_recomputed':False})
    accepted=HERE/'verified_snapshot/accepted_designs.csv'
    with accepted.open(encoding='utf-8-sig',newline='') as h:rows=list(csv.DictReader(h))
    checks=[];curves=[]
    for row in rows:
        path=Path(row['curve_path']);curves.append(path)
        assert sha(path)==row['curve_sha256']
        data=read_csv_rows(path);ensure_db_columns(data)
        old=feature_based_impedances(data,50,6.85);new=analyze(path,6.85)
        fixed=old['preferred_fixed_input_f0'];p=old['product_zm_zs'];r=fixed['zm_prime']/fixed['zs_prime']
        assert abs(p-new['formal_response_PR']['P_response'])<1e-9
        assert abs(r-new['R_side_poles_diagnostic'])<1e-9
        assert abs(r-float(row['R_side_poles_fixed_f0']))<1e-9
        checks.append({'case':row['case'],'old_feature_P':p,'current_P':new['formal_response_PR']['P_response'],
            'old_fixed_f0_R':r,'current_R_side_poles_fixed_f0':new['R_side_poles_diagnostic'],
            'R_BW':new['formal_response_PR']['R_BW'],'formula_match':True})
    sources=[selection,index,accepted,Path(__file__),WORK/'fit_modified_impedances.py',
        WORK/'output/folded_simulation_plan_20260920/response_metrics.py',
        Path('F:/2026-1/circuit/mcp/scripts/run_folded_r8.py'),
        Path('F:/2026-1/circuit/weekpost/prepare_txline_training_dataset_1000.py'),*curves]
    report={'created_at_utc':datetime.now(timezone.utc).isoformat(),'uniform_index_rows':len(uniform),
        'seed_label_checks':seeds,'accepted_folded_formula_checks':checks,
        'f0_GHz':6.85,'R_formula':'(P-1)/(P*mean(abs(cot(pi*f_outer/(2*f0))))**2)',
        'limitations':['Eight historical folded labels were copied from uniform seeds, not extracted from folded responses.',
            'Original uniform seed curves are not recomputed; row-level extraction provenance for all 1000 entries is not verified.',
            'TXLINE initialization code uses er=3.38; current HFSS uses er=3.55. These are not interchangeable physical impedance labels.',
            'This R uses response features; it is not a geometry-only predictor. No independent bandwidth law is fitted.',
            'Current two accepted configurations still require width-corner mode checks.'],
        'source_hashes':{str(p):sha(p) for p in sources}}
    dest=HERE/'uniform_pr_definition_audit.json';dest.write_text(json.dumps(report,indent=2),encoding='utf-8')
    table='\n'.join(f"| {x['case']} | {x['current_P']:.6f} | {x['current_R_side_poles_fixed_f0']:.6f} | {x['R_BW']:.6f} |" for x in checks)
    text=f'''# 均匀模型与弯折模型的 P、R 比较口径

核验时间：{report['created_at_utc']}。本记录核验参数定义，不宣称找到新带宽规律。

旧均匀索引实际为1000行。历史8个弯折种子的 P_eff、R_eff 是复制均匀种子标签；8行均与索引中 Zm′、Zs′ 的乘积/2500及比值吻合，不能当作弯折模型重新提取的结果。旧种子原始曲线尚未复算，1000行的逐行提取来源也尚未全部核验。

旧特征法先从两个 S11 峰的线性幅度均值求 P，再以固定 f0=6.85 GHz 和两侧外谷计算 u=mean(abs(cot(πf/(2f0))))，得到 R=(P−1)/(P u²)。S21≥−2 dB 用于筛选通带；BW 数值没有用于这个 R 的反算。当前冻结提取器已有同一公式，现将其作为 R_side_poles_fixed_f0 导出。两个已通过配置的曲线经旧函数独立复算，P、R 均在1e−9内一致。

| 配置 | P_response | R_side_poles_fixed_f0 | R_BW |
|---|---:|---:|---:|
{table}

R_zero_mean 使用本点中央谷频率、分别算左右 R 再平均，属于另一口径；R_BW 使用带宽反算，不能用它与该带宽的相关性称作独立规律。这些响应参数均不等于从几何独立预测的物理阻抗。TXLINE初值脚本还使用 εr=3.38，而本轮HFSS为3.55，不能直接混为同一材料口径。

接下来先完成24配置及角点验收，再在相同宽度和折臂配对中检查 BW 与上述 S11 特征 R、P 的关系及残差。首批 g=0.55 mm 固定，不能单独确定间隔规律；目前两个配置不足以支持一般关系。

机器可复查数据与文件哈希：[uniform_pr_definition_audit.json](uniform_pr_definition_audit.json)。
'''
    (HERE/'均匀与弯折PR口径核验.md').write_text(text,encoding='utf-8')
    print(json.dumps({'uniform_rows':len(uniform),'seed_matches':len(seeds),'curve_formula_matches':len(checks),'report':str(dest)}))


if __name__=='__main__':main()

def read(p,default=None):
    try:return json.loads(Path(p).read_text(encoding='utf-8-sig'))
    except (OSError,ValueError):return default
def atomic(p,text):
    tmp=p.with_suffix(p.suffix+'.tmp');tmp.write_text(text,encoding='utf-8');os.replace(tmp,p)
def link(p,title=None):return f'[{title or Path(p).name}]({Path(p).as_posix()})'
def clean(x):return str(x).replace('|','/').replace('\n',' ')[:200]

def recovery_accepted(path):

    if 'local_repair' not in path.parts:return False
    case_component=path.parts[path.parts.index('local_repair')+1]

    if match:return bool(read(ROOT/match[1]/match[2]/'recovery_acceptance.json',{}).get('approved'))
    if case_component=='D04G05':return bool(read(ROOT/'D04G05/i00/recovery_acceptance.json',{}).get('approved'))
    return False

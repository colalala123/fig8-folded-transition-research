"""Finish the live B0 copy check, then serially verify the rebuilt collector.

Stops with collector_review.json; never freezes production or dispatches pilot.
"""
import asyncio,csv,json,os,sys,time,traceback
from pathlib import Path
HERE=Path(__file__).resolve().parent
ROOT=Path('J:/f8_261007')
sys.path.insert(0,str(HERE.parent/'fig8_handoff_20261002'))
from pilot_worker import now,sha,write_json
from pilot_queue import compare
from pilot_supervisor import aux_compare,verify_boundaries
from assemble_segments import assemble
from export_segment_auxiliary import export
from complete_reference_evidence import common_fields

def read(p):return json.loads(Path(p).read_text(encoding='utf-8'))
def status(**kw):write_json(ROOT/'collector_workflow_state.json',{'pid':os.getpid(),'updated_at_utc':now(),**kw})
def decision(action,next_action):
    write_json(ROOT/'execution_decision.json',{'current_action':action,'next_action':next_action,'selected_modes':2})

def finalize(run):
    result=run/'collector_case_review.json'
    if result.exists():
        saved=read(result)
        for file,digest in saved['evidence_hashes'].items():assert sha(file)==digest
        return saved
    s=read(run/'state.json');assert s['status']=='segments_exported_review_required'
    numerical=assemble(run,2)
    mapping={k:s.get('sweep_names',{}).get(k,k) for k in ['Low','Bridge','High','Endpoint']}
    export(s['project'],s['setup'],mapping,run/'auxiliary',s['port'],2)
    data={}
    for k in mapping:
        with (run/'auxiliary'/f'{k}_auxiliary.csv').open(encoding='utf-8') as h:data[k]={round(float(x['Freq_GHz']),2):x for x in csv.DictReader(h)}
    rows=[]
    for i in range(1,1501):
        f=round(i*.01,2);k='Bridge' if 6.96<=f<=7.01 else 'Endpoint' if i==1500 else 'Low' if f<6.98 else 'High';rows.append(data[k][f])
    with (run/'mode_impedance_gamma.csv').open('w',newline='',encoding='utf-8') as h:
        w=csv.DictWriter(h,fieldnames=rows[0]);w.writeheader();w.writerows(rows)
    write_json(run/'manifest.json',{'status':'completed' if numerical['passed'] else 'numerical_review_required','project':s['project'],'source':s['source'],'source_sha256':s['source_sha256'],'source_unchanged':s['source_unchanged'],'modes':2,'port_settings':s['port_settings'],'geometry':s.get('geometry'),'curve_sha256':sha(run/'sparams.csv')})
    assert numerical['passed'],'Numerical review failed; preserve results for repair'
    old=Path('F:/hfss_f8x_261002/M_B0_2')
    out={'numerical':numerical,'original_protocol_comparison':compare(old,run,'collector'),'fields':common_fields(old,run),'auxiliary':aux_compare(old,run),'boundaries':verify_boundaries(run)}
    out['passed']=all(v['passed'] for v in out.values())
    files=[run/x for x in ['sparams.csv','response_metrics.json','state.json','manifest.json','mode_impedance_gamma.csv','numerical_review.json']]
    out['evidence_hashes']={str(p):sha(p) for p in files}
    write_json(result,out);return out

async def main():
    raise RuntimeError('Single15 collector retired after direct two/three-mode failure. Use the reviewed Broadband workflow; preserve completed trials.')
    reference=read(ROOT/'reference_review.json')
    assert reference['passed'] and reference['recommended_modes']==2
    status(status='waiting_for_B0_copy')
    while True:
        path=ROOT/'B0M2/state.json'
        if path.exists():
            state=read(path)
            if state['status']=='review_required':raise RuntimeError('B0 copy solver needs review; no further dispatch')
            if state['status']=='segments_exported_review_required' and not (ROOT/'protocol.lock').exists():break
        await asyncio.sleep(30)
    copy=finalize(ROOT/'B0M2');assert copy['passed'],'Copy collector comparison failed'
    status(status='B0_copy_passed_preparing_rebuild')
    decision('B0二模复制收集器已通过，正在验证同尺寸几何重建','重建对照通过后复核并冻结生产协议；然后监督前几个配置')
    import run_protocol_tf3 as backend
    backend.RUN=ROOT/'B0R2';backend.PROJECT=backend.RUN/'B0R2.aedt';backend.STATE=backend.RUN/'state.json'
    backend.SOURCE=Path('F:/hfss_f8_261002/B0/B0.aedt');backend.SEED=ROOT/'no_mesh_seed.aedt';backend.MODE_COUNT=2
    backend.REBUILD_GEOMETRY=read(backend.SOURCE.parent/'manifest.json')['geometry']
    backend.RUN_SCOPE='Authorized B0 geometry-rebuild collector check only, not a pilot phase iteration'
    if not backend.STATE.exists():await backend.main()
    else:
        s=read(backend.STATE)
        if s['status']!='segments_exported_review_required':raise RuntimeError('Existing rebuild requires recovery; do not restart blindly')
    rebuild=finalize(backend.RUN)
    matched=compare(ROOT/'B0M2',ROOT/'B0R2','collector')
    out={'copy':copy,'rebuild':rebuild,'copy_vs_rebuild':matched,'passed':copy['passed'] and rebuild['passed'] and matched['passed'],'production_dispatch_started':False,'reviewed_at_utc':now()}
    files=[ROOT/'B0M2/collector_case_review.json',ROOT/'B0R2/collector_case_review.json',ROOT/'reference_review.json']
    out['evidence_hashes']={str(p):sha(p) for p in files}
    write_json(ROOT/'collector_review.json',out)
    assert out['passed'],'Rebuilt collector comparison failed'
    status(status='collector_complete_production_review_required')
    decision('复制/重建收集器均已通过，等待生产协议最终复核','冻结协议、审查历史结果，再监督首几个配置并放行剩余队列')

if __name__=='__main__':
    try:asyncio.run(main())
    except Exception:
        status(status='review_required',error=traceback.format_exc());raise

"""Review T_F3 segmented verification without initiating another solve."""
import csv,json,sys,re
from pathlib import Path
import numpy as np
from audit_existing import HERE,OLD,read,write,sha,touchstone,curve,analyze
sys.path.insert(0,str(HERE.parent/'fig8_handoff_20261002'))
import pilot_queue as q
from port_evidence import collect,signature
RUN=Path('J:/f8_261007/TF3M3')
SOURCES=['Low','Bridge','High','Endpoint']
def review():
    state=read(RUN/'state.json')
    assert all(state['segments'].get(x,{}).get('status')=='exported' for x in SOURCES),'All segment exports required'
    from assemble_segments import assemble
    assembly=assemble(RUN,3)
    target=np.arange(1,1501)*.01;audit={'source_unchanged':sha(state['source'])==state['source_sha256'],'normalizations':{},'comparisons':[]}
    for label in ['native','50ohm']:
        datasets={k:touchstone(RUN/f'{label}_{k}.s6p') for k in SOURCES}
        result=[];provenance=[]
        for ff in target:
            name='Bridge' if 6.96-1e-8<=ff<=7.01+1e-8 else 'Endpoint' if abs(ff-15)<1e-8 else 'Low' if ff<6.98 else 'High'
            f,s=datasets[name];inds=np.flatnonzero(abs(f-ff)<1e-8);assert len(inds)==1
            result.append(s[inds[0]]);provenance.append(name)
        s=np.array(result);assert np.isfinite(s).all()
        p=RUN/(label+'_sparams.csv');curve(p,target,s);m=analyze(p)
        m['validity']['solver_convergence_verified']=True
        if not m['validity'].get('equal_ripple'):m['formal_response_PR']=None
        write(RUN/(label+'_metrics.json'),m)
        entry={'points':len(target),'finite':True,'curve_sha256':sha(p),'sources':{k:provenance.count(k) for k in SOURCES}}
        if label=='50ohm':
            sv=np.linalg.svd(s,compute_uv=False)[:,0];entry.update(max_sv=float(max(sv)),passivity_violations=int(sum(sv>1.001)),max_reciprocity=float(abs(s-s.transpose(0,2,1)).max()))
        overlaps=[]
        for left,right in [('Low','Bridge'),('High','Bridge'),('High','Endpoint')]:
            fa,sa=datasets[left];fb,sb=datasets[right]
            for j,ff in enumerate(fb):
                ix=np.flatnonzero(abs(fa-ff)<1e-8)
                if len(ix):
                    a,b=sa[ix[0]],sb[j];overlaps.append({'GHz':float(ff),'left':left,'right':right,'complex_max':float(abs(a-b).max()),'fundamental_max_abs_magnitude_difference':float(max(abs(abs(a[i,jj])-abs(b[i,jj])) for i,jj in [(0,0),(3,0)]))})
        entry['overlaps']=overlaps;entry['fundamental_overlap_passed']=all(x['fundamental_max_abs_magnitude_difference']<=.02 for x in overlaps)
        audit['normalizations'][label]=entry
        if label=='native':
            (RUN/'sparams.csv').write_bytes(p.read_bytes());write(RUN/'response_metrics.json',m)
    audit['comparisons']=[q.compare(OLD/'M_T_F3_2',RUN,'stable'),q.compare(q.OLD/'T_F3',RUN,'single')]
    # Actual exported labels, not matrix size alone, establish port indexing.
    maps=[]
    for seg in SOURCES:
        header=(RUN/f'50ohm_{seg}.s6p').read_text().splitlines()[:20]
        mapping=[x.strip() for x in header if x.startswith('! Port[')]
        assert mapping==[f'! Port[{i+1}] = {p}:{mode}' for i,(p,mode) in enumerate([(1,1),(1,2),(1,3),(2,1),(2,2),(2,3)])]
        maps.append({'segment':seg,'mapping':mapping})
    audit['mode_mapping']=maps
    # Save an adapter manifest for the existing field fingerprint collector.
    write(RUN/'manifest.json',{'status':'completed','project':state['project'],'port_settings':state['port_settings'],'source':state['source'],'source_sha256':state['source_sha256'],'source_unchanged':audit['source_unchanged'],'scope':'completed numerical sweeps; protocol decision in protocol_review.json'})
    try:
        ref=collect(OLD/'M_T_F3_2');new=collect(RUN);fields=[]
        for rr in ref:
            nn=[x for x in new if abs(x['frequency_GHz']-rr['frequency_GHz'])<1e-8]
            if not nn:continue
            for port in ['1','2']:
                a=next(x for x in rr['fields'] if x['port']==port and x['mode']==1);vx=signature(a)
                scores={}
                for b in nn[0]['fields']:
                    if b['port']!=port:continue
                    assert np.allclose(a['grid_bounds_native_units'],b['grid_bounds_native_units'])
                    vy=signature(b);scores[b['mode']]=float(vx@vy/(np.linalg.norm(vx)*np.linalg.norm(vy)))
                corr=scores[1];margin=corr-max(abs(v) for k,v in scores.items() if k!=1)
                fields.append({'GHz':rr['frequency_GHz'],'port':port,'scores':scores,'passed':corr>=.98 and margin>=.1})
        audit['fields']={'evidence':fields,'passed':len(fields)>=2 and all(x['passed'] for x in fields),'scope':'common saved frequencies only; full-band auxiliary review still required'}
    except Exception as e:audit['fields']={'passed':False,'error':repr(e)}
    audit['assembly']=assembly
    audit['numerical_gate_passed']=assembly['passed'] and audit['source_unchanged'] and audit['normalizations']['50ohm']['passivity_violations']==0 and all(x['fundamental_overlap_passed'] for x in audit['normalizations'].values()) and audit['comparisons'][0]['passed']
    audit['production_release']=False
    audit['release_note']='No automatic release: both reference protocols, auxiliary quantities, collector and supervised configurations still require review.'
    write(RUN/'protocol_review.json',audit);print(json.dumps(audit,indent=2))
if __name__=='__main__':review()

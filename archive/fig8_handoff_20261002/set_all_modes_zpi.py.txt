import shutil
from pathlib import Path
from ansys.aedt.core import Hfss

root = Path(r"F:/hfss_f8x_261002/R_B0_rebuild_max500")
p = root / "R_B0_rebuild_max500.aedt"
shutil.copy2(p, root / "R_B0_rebuild_max500.before_all_modes_zpi.aedt")
app = Hfss(project=str(p), design="Wideband vertical transition", version="2024.2", machine="localhost", port=7393, non_graphical=False, new_desktop=False, close_on_exit=False)
try:
    changed = {}
    for name in ["1", "2"]:
        b = next(x for x in app.boundaries if x.name == name)
        before = b.props["Modes"]["Mode3"].get("CharImp")
        b.props["Modes"]["Mode3"]["CharImp"] = "Zpi"
        b.update()
        changed[name] = {"before": before, "after": b.props["Modes"]["Mode3"].get("CharImp")}
    app.save_project()
    print(changed, flush=True)
finally:
    app.release_desktop(close_projects=False, close_desktop=False)

"""Eight direct diagnostic frequencies using preserved converged mesh, via visible MCP."""
import asyncio,shutil,json,traceback
from pathlib import Path
import psutil
from ansys.aedt.core import Hfss
from mcp_call import call
from pilot_worker import now,write_json,sha
ROOT=Path('F:/hfss_f8x_261002');SRC=ROOT/'M_B0_3w16';RUN=ROOT/'R_B0_direct8';NAME='M_B0_3w16';PORT=12704
FREQ=[.01,.1,3.24,6.73,9.45,10.01,14.11,15.]
def attach():
    return Hfss(project=str(RUN/(NAME+'.aedt')),design='Wideband vertical transition',version='2024.2',machine='localhost',port=PORT,non_graphical=False,new_desktop=False,close_on_exit=False)
async def main():
    if any((p.info['name'] or '').lower()=='hf3d.exe' for p in psutil.process_iter(['name'])):raise RuntimeError('Solver active')
    if RUN.exists():raise RuntimeError('Preserve existing trial')
    await call('close_projects',{'project_names':[NAME],'save':True,'port':PORT,'timeout':120})
    RUN.mkdir();shutil.copy2(SRC/(NAME+'.aedt'),RUN/(NAME+'.aedt'));shutil.copytree(SRC/(NAME+'.aedtresults'),RUN/(NAME+'.aedtresults'))
    m={'status':'preparing','source':str(SRC),'source_sha256':sha(SRC/(NAME+'.aedt')),'diagnostic_only':True,'cores':16,'tasks':1,'frequencies_GHz':FREQ,'started_at_utc':now()}
    app=attach()
    try:
        assert Path(app.project_path).resolve()==RUN.resolve()
        setup=app.get_setup('Setup19');old=next(s for s in setup.sweeps if s.name=='Sweep1')
        old.props['IsEnabled']=False;assert old.update()
        sw=setup.create_single_point_sweep(unit='GHz',freq=FREQ.copy(),name='DirectCheck',save_single_field=False,save_fields=False)
        assert sw
        m['sweep']=dict(sw.props);m['setup']=dict(setup.props);m['messages_before']=list(app.odesktop.GetMessages(NAME,app.design_name,0));app.save_project()
    finally:app.release_desktop(close_projects=False,close_desktop=False)
    write_json(RUN/'repair_manifest.json',m)
    m['mcp_start']=await call('start_analysis',{'project_name':NAME,'design_name':'Wideband vertical transition','setup_name':'Setup19','port':PORT,'concurrent_tasks':1,'cores_per_task':16,'blocking':False,'timeout':120})
    assert m['mcp_start'].get('started');m['status']='solving';write_json(RUN/'repair_manifest.json',m)
    while True:
        await asyncio.sleep(20)
        st=await call('get_analysis_status',{'project_name':NAME,'design_name':'Wideband vertical transition','port':PORT,'timeout':120})
        if not st.get('running'):break
    app=attach()
    try:
        m['messages']=list(app.odesktop.GetMessages(NAME,app.design_name,0))
        for norm in (False,True):
            path=RUN/('hfss_50ohm.s6p' if norm else 'hfss_native.s6p')
            assert app.export_touchstone(setup='Setup19',sweep='DirectCheck',output_file=str(path),renormalization=norm,impedance=50,gamma_impedance_comments=True)
        m['source_unchanged']=sha(SRC/(NAME+'.aedt'))==m['source_sha256'];m['status']='exported_needs_review';m['finished_at_utc']=now();app.save_project();write_json(RUN/'repair_manifest.json',m)
    finally:app.release_desktop(close_projects=False,close_desktop=False)
if __name__=='__main__':
    try:asyncio.run(main())
    except Exception:
        if RUN.exists():(RUN/'error.txt').write_text(traceback.format_exc(),encoding='utf-8')
        raise

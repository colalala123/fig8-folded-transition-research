import json
from pathlib import Path
from ansys.aedt.core import Hfss

R = Path(r"F:\hfss_f8x_261align2")
P = R / "R_tri_linealign2.aedt"
app = Hfss(project=str(P), design="Wideband vertical transition", version="2024.2", machine="localhost", port=7990, non_graphical=False, new_desktop=False, close_on_exit=False)
try:
    setup = app.get_setup("TriDiagE")
    for old in setup.sweeps:
        old.props["IsEnabled"] = False
        old.update()
    sweep = setup.add_sweep("SweepLineAlign2NoPass", "Interpolating")
    sweep.props.update({
        "RangeType": "LinearStep", "RangeStart": "0.01GHz", "RangeEnd": "15GHz", "RangeStep": "0.01GHz",
        "InterpTolerance": 0.5, "InterpMaxSolns": 500, "InterpMinSolns": 0, "InterpMinSubranges": 1,
        "InterpUseS": True, "InterpUsePortImped": False, "InterpUsePropConst": False,
        "UseDerivativeConvergence": False, "InterpDerivTolerance": 0.2, "UseFullBasis": True,
        "EnforcePassivity": False, "PassivityErrorTolerance": 0.0001, "EnforceCausality": False,
        "SMatrixOnlySolveMode": "Auto", "SMatrixOnlySolveAbove": "15GHz",
        "SaveFields": True, "SaveRadFields": False,
    })
    sweep.update()
    app.save_project()
    print(json.dumps({"status": "prepared", "sweep": "SweepLineAlign2NoPass", "props": json.loads(json.dumps(sweep.props, default=str))}, indent=2))
finally:
    app.release_desktop(close_projects=False, close_desktop=False)

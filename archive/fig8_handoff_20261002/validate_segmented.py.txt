import json
from pathlib import Path
import numpy as np

R = Path(r"F:\hfss_f8x_261002\R_tri_diagC")

def parse(path):
    vals = []
    unit = 1.0
    fmt = "RI"
    for line in Path(path).read_text(errors="ignore").splitlines():
        line = line.strip()
        if not line or line.startswith("!"):
            continue
        if line.startswith("#"):
            t = line.upper().split()
            unit = {"HZ": 1, "KHZ": 1e3, "MHZ": 1e6, "GHZ": 1e9}.get(next((x for x in t if x in ("HZ", "KHZ", "MHZ", "GHZ")), "GHZ"), 1e9)
            fmt = next((x for x in ("RI", "MA", "DB") if x in t), "RI")
            continue
        vals.extend(float(x.replace("D", "E")) for x in line.split())
    stride = 73
    chunks = [vals[i : i + stride] for i in range(0, len(vals) - stride + 1, stride)]
    f = np.array([c[0] * unit / 1e9 for c in chunks])
    s = np.zeros((len(chunks), 6, 6), complex)
    for k, c in enumerate(chunks):
        x = np.array(c[1:]).reshape(36, 2)
        z = x[:, 0] * np.exp(1j * np.deg2rad(x[:, 1])) if fmt == "MA" else x[:, 0] + 1j * x[:, 1]
        s[k] = z.reshape(6, 6)
    return f, s

out = {}
for name in ["50ohm_segmented_repaired.s6p", "native_segmented_repaired.s6p"]:
    f, s = parse(R / name)
    sv = np.linalg.svd(s, compute_uv=False)
    maxsv = sv.max(axis=1)
    d = np.max(np.abs(s[1:] - s[:-1]), axis=(1, 2))
    out[name] = {
        "points": int(len(f)), "start_GHz": float(f[0]), "end_GHz": float(f[-1]),
        "max_singular_value": float(maxsv.max()), "max_sv_freq_GHz": float(f[maxsv.argmax()]),
        "min_singular_value": float(sv.min()), "passivity_violations": int((maxsv > 1 + 1e-9).sum()),
        "max_reciprocity_error": float(np.max(np.abs(s - np.transpose(s, (0, 2, 1))))),
        "finite": bool(np.isfinite(s).all()),
        "max_adjacent_matrix_step": float(d.max()), "largest_step_freq_GHz": float(f[1:][d.argmax()]),
    }
print(json.dumps(out, indent=2))
(R / "metrics_segmented_repaired.json").write_text(json.dumps(out, indent=2), encoding="utf-8")

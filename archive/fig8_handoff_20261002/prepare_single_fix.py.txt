import json, hashlib, time, sys
from pathlib import Path
from ansys.aedt.core import Hfss
sys.path.insert(0, r"${WORKSPACE}\output\fig8_handoff_20261002")
from pilot_worker import set_modes

ROOT = Path(r"F:\hfss_f8x_261002\R_B0_single_fix")
PROJECT = ROOT / "R_B0_single_fix.aedt"
MANIFEST = ROOT / "single_fix_manifest.json"
PORT = 12704

def sha(path):
    h=hashlib.sha256()
    with path.open("rb") as f:
        for b in iter(lambda:f.read(1<<20),b""): h.update(b)
    return h.hexdigest()

before=sha(PROJECT)
app=Hfss(project=str(PROJECT),design="Wideband vertical transition",version="2024.2",machine="localhost",port=PORT,non_graphical=False,new_desktop=False,close_on_exit=False)
try:
    mode_change=set_modes(app,1)
    setup=app.create_setup(name="ModalSingle",setup_type="HFSSDriven")
    setup.props.update({"SolveType":"Single","Frequency":"15GHz","MaxDeltaS":0.005,
                        "UseMatrixConv":False,"MaximumPasses":40,"MinimumPasses":6,
                        "MinimumConvergedPasses":2,"PercentRefinement":30,"BasisOrder":1,
                        "PortAccuracy":2,"DrivenSolverType":"Direct Solver","SaveAnyFields":True,
                        "SaveRadFieldsOnly":False,"MeshLink":{"ImportMesh":False}})
    setup.update()
    sweep=setup.create_linear_step_sweep(unit="GHz",start_frequency=0.01,stop_frequency=15.0,
                                         step_size=0.01,name="SweepSingle",save_fields=True,
                                         save_rad_fields=False,sweep_type="Interpolating")
    sweep.props.update({"IsEnabled":True,"RangeType":"LinearStep","RangeStart":"0.01GHz",
                        "RangeEnd":"15GHz","RangeStep":"0.01GHz","Type":"Interpolating",
                        "InterpTolerance":0.5,"InterpMaxSolns":500,"InterpMinSolns":0,
                        "InterpMinSubranges":1,"InterpUseS":True,"InterpUsePortImped":False,
                        "InterpUsePropConst":True,"UseDerivativeConvergence":False,
                        "InterpDerivTolerance":0.2,"UseFullBasis":True,"EnforcePassivity":True,
                        "PassivityErrorTolerance":0.0001,"EnforceCausality":False,
                        "SMatrixOnlySolveMode":"Manual","SMatrixOnlySolveAbove":"15GHz",
                        "SaveFields":True,"SaveRadFields":False})
    sweep.update(); app.save_project()
    data={"status":"prepared","created_at_utc":time.strftime('%Y-%m-%dT%H:%M:%SZ',time.gmtime()),
          "project":str(PROJECT),"project_sha256_before":before,"port":PORT,"cores":16,"tasks":1,
          "diagnostic_only":True,"mode_change":mode_change,"setup":"ModalSingle","sweep":"SweepSingle",
          "range_GHz":[0.01,15.0,0.01],"change":"one propagating mode per wave port; Manual S-matrix; no port impedance basis",
          "setup_props":json.loads(json.dumps(setup.props,default=str)),"sweep_props":json.loads(json.dumps(sweep.props,default=str)),
          "messages_before":list(app.odesktop.GetMessages(app.project_name,app.design_name,0))}
    MANIFEST.write_text(json.dumps(data,indent=2,default=str),encoding="utf-8"); print(json.dumps(data,indent=2,default=str))
finally: app.release_desktop(close_projects=False,close_desktop=False)

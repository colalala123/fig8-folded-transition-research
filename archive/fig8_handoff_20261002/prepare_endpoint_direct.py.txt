import json
from pathlib import Path
from ansys.aedt.core import Hfss

R = Path(r"F:\hfss_f8x_261002\R_tri_diagC")
P = R / "R_tri_diagC.aedt"
app = Hfss(project=str(P), design="Wideband vertical transition", version="2024.2", machine="localhost", port=7990, non_graphical=False, new_desktop=False, close_on_exit=False)
try:
    setup = app.get_setup("TriDiagE")
    for old in setup.sweeps:
        old.props["IsEnabled"] = False
        old.update()
    sweep = setup.add_sweep("SweepEndpointDirect", "Discrete")
    sweep.props.update({"RangeType": "LinearCount", "RangeStart": "15GHz", "RangeEnd": "15GHz", "RangeCount": 1, "SaveFields": False, "SaveRadFields": False})
    sweep.update(); app.save_project()
    print(json.dumps({"status": "prepared", "sweep": "SweepEndpointDirect", "props": json.loads(json.dumps(sweep.props, default=str))}, indent=2))
finally:
    app.release_desktop(close_projects=False, close_desktop=False)

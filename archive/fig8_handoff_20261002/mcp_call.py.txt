"""Call the local AEDT MCP over its real stdio transport."""
import asyncio,json,os,sys
from pathlib import Path
from datetime import timedelta
from contextlib import asynccontextmanager
from mcp import ClientSession,StdioServerParameters
from mcp.client.stdio import stdio_client

REPO=Path('F:/2026-1/circuit/mcp')
@asynccontextmanager
async def connected_client():
    env={**os.environ,'AEDT_VERSION':'2024.2','AEDT_INSTALL_DIR':'D:/Program Files/AnsysEM/v242/Win64',
         'AEDT_WORKER_TIMEOUT':'120','AEDT_LAUNCH_TIMEOUT':'180'}
    params=StdioServerParameters(command=sys.executable,args=[str(REPO/'mcp_server.py')],cwd=str(REPO),env=env)
    async with stdio_client(params) as streams:
        async with ClientSession(*streams,read_timeout_seconds=timedelta(seconds=240)) as session:
            await session.initialize()
            yield session

async def session_call(session,name,arguments):
    if name=='list_tools':return (await session.list_tools()).model_dump(mode='json')
    result=await session.call_tool(name,arguments)
    if result.isError:raise RuntimeError(result.model_dump_json())
    if result.structuredContent:return result.structuredContent
    return json.loads(next(c.text for c in result.content if c.type=='text'))

async def call(name,arguments):
    async with connected_client() as session:return await session_call(session,name,arguments)

if __name__=='__main__':
    args=json.loads(Path(sys.argv[2]).read_text(encoding='utf-8')) if len(sys.argv)>2 else {}
    result=asyncio.run(call(sys.argv[1],args))
    if len(sys.argv)>3:Path(sys.argv[3]).write_text(json.dumps(result,indent=2),encoding='utf-8')
    print(json.dumps(result,indent=2),flush=True)

import ast
from pathlib import Path
import phase_control as pc


def main():
    root = pc.HERE / 'length_tuning'
    assert not root.exists(), 'Inspect existing segment; never overwrite frozen inputs'
    ledger = pc.read(pc.HERE/'budget_ledger.json')
    assert ledger['new_geometry_reserved'] == 18 and len(ledger['tasks']) == 18
    assert all(t['status']=='reviewed' for t in ledger['tasks'])
    rows = pc.read(pc.HERE/'plan.json')['cases']
    frozen = {}; failed = []; summary = []
    for row in rows:
        run = pc.BASE/row['case']/'g01_initial'
        ack=pc.read(run/'receiver_ack.json'); event=pc.read(run/'completion_event.json')
        assert ack['event_sha256']==pc.sha(run/'completion_event.json')
        assert ack['verification_sha256']==pc.sha(run/'receiver_verification.json')
        assert ack['operation_success'] and ack['visual_review_completed']
        for item in event['artifacts']: assert pc.sha(run/item['path'])==item['sha256']
        spec=pc.read(run/'input.json')
        for file,digest in spec['parent_evidence_hashes'].items(): assert pc.sha(file)==digest
        result=pc.read(run/'finish24_result.json'); m=result['metrics']; t=pc.read(run/'target20_metrics.json')
        assert result['geometry']==row['geometry'] and result['design_passed']==ack['design_passed']
        if not result['design_passed']: failed.append(row['case'])
        for name in ['input.json','receiver_ack.json','receiver_verification.json','completion_event.json','response_metrics.json','target20_metrics.json','sparams.csv','state.json']:
            frozen[str(run/name)] = pc.sha(run/name)
        summary.append(dict(case=row['case'],P=row['target_P'],R=row['target_R'],a_mm=row['a_mm'],
            actual_elements=result['actual_elements'],baseline_passed=result['design_passed'],
            center_GHz=m['central_valley_GHz'],peak_difference_dB=m['peak_difference_dB'],
            transmission_BW_GHz=m['passband_2dB']['BW_GHz'],target20=t,
            frozen_prediction_BW_GHz=row['predicted_BW_GHz']))
    assert failed==['P01G05','P04G05'], 'Review unexpected failed-case set'
    frozen[str(pc.HERE/'budget_ledger.json')]=pc.sha(pc.HERE/'budget_ledger.json')
    frozen[str(pc.HERE/'authorization.json')]=pc.sha(pc.HERE/'authorization.json')
    root.mkdir(); (root/'hook').mkdir()
    provenance={}
    def copy(name, transform=None):
        source=pc.HERE/name; text=source.read_text(encoding='utf-8')
        if transform: text=transform(text)
        target=root/name; target.parent.mkdir(parents=True,exist_ok=True);target.write_text(text,encoding='utf-8')
        provenance[name]=dict(source=str(source),source_sha256=pc.sha(source),local_sha256=pc.sha(target))
    for name in ['phase_geometry_worker.py','case_reference.py','reference_validation.json','prepared_settings.py',
                 'target_metrics.py','policy.json','mesh_policy_addendum_v2.json','desktop_start.json','aedt_identity.json',
                 'aedt_session_guard.py','phase_publish.py','phase_heartbeat.py','phase_heartbeat_transport.py',
                 'review_phase_geometry.py','supervise_start.py']:
        transform=(lambda s:s.replace("HERE.parent/'fig8_speed_study_20261021'", "ORIGINAL.parent/'fig8_speed_study_20261021'")) if name=='phase_geometry_worker.py' else None
        copy(name,transform)
    control=(pc.HERE/'phase_control.py').read_text(encoding='utf-8')
    control=control.replace("ORIGINAL=HERE.parent/'fig8_finish24_20261023'", "PARENT=HERE.parent\nORIGINAL=PARENT.parent/'fig8_finish24_20261023'")
    control=control.replace("HOOK=HERE.parent/'completion-hook-portable-1.0.0'", "HOOK=PARENT.parent/'completion-hook-portable-1.0.0'")
    begin=control.index('ORDER=');end=control.index('\n',begin)
    control=control[:begin]+'ORDER='+repr(failed)+control[end:]
    (root/'phase_control.py').write_text(control,encoding='utf-8')
    # New scoped checks; original worker/context/inputs remain immutable.
    (root/'phase_geometry_context.py').write_text(CONTEXT,encoding='utf-8')
    ack=(pc.HERE/'acknowledge_review.py').read_text(encoding='utf-8')
    ack=ack.replace("assert case in pc.ORDER and run==(pc.BASE/case/'g01_initial').resolve()", "from phase_geometry_context import job_name\n    assert case in pc.ORDER and run==(pc.BASE/case/job_name(spec['new_phase_geometry_ordinal'])).resolve()")
    begin=ack.index("    gate=pc.HERE/");end=ack.index("    print(",begin)
    ack=ack[:begin]+ack[end:]
    (root/'acknowledge_review.py').write_text(ack,encoding='utf-8')
    pub=(pc.HERE/'phase_geometry_publish.py').read_text(encoding='utf-8')
    begin=pub.index('def prompt(');end=pub.index('async def publish',begin)
    pub=pub[:begin]+PUBLISH_PROMPT+'\n'+pub[end:]
    pub=pub.replace('PR18 initial validation after explicit launch instruction','PR18 bounded length tuning authorized within first three steps')
    (root/'phase_geometry_publish.py').write_text(pub,encoding='utf-8')
    (root/'advance.py').write_text(ADVANCE,encoding='utf-8')
    pc.write(root/'plan.json',dict(stage='PR18_lengths',cases=[r for r in rows if r['case'] in failed]))
    pc.write(root/'parent_evidence.json',dict(hashes=frozen,initial_geometries=18))
    pc.write(root/'authorization.json',dict(stage='PR18_lengths',cases=failed,execution_authorized=True,
        human_instruction='你现在把前三步做了，补点仿真用记得用hook',extra_geometry_cap=6,
        per_case_extra_cap=3,global_geometry_cap=30,shared_extra_cap=12,
        rationale='Only two baseline balance failures; one measured two-probe/one-candidate cycle each, stop on passing probe. No -20dB width tuning.'))
    pc.write(root/'budget_ledger.json',dict(new_geometry_reserved=0,new_cached_tasks_reserved=0,tasks=[]))
    cfg=pc.read(pc.HERE/'hook/config.json');cfg['state_dir']=str(root/'hook/state');pc.write(root/'hook/config.json',cfg)
    (root/'CONTINUATION.md').write_text(CONTINUATION,encoding='utf-8')
    pc.write(root/'implementation_provenance.json',provenance)
    pc.write(root/'active.json',dict(stage='configured_not_launched',at=pc.now()))
    pc.write(pc.HERE/'initial18_summary.json',dict(at=pc.now(),initial_completed=18,baseline_passed=16,
        tuning_cases=failed,peak20_positive=sum(s['target20']['both_inner_peaks_le_minus20'] for s in summary),
        final_direct_review_completed=False,cases=summary))
    lines=['# PR18 初始18点汇总（非最终验收）','',
        '18/18求解完成，16项基础调谐通过；P01G05、P04G05峰差分别约0.618、0.624 dB，需有界长度补偿。',
        '两内峰≤−20 dB共6项；其余保留为科学负例。所有项最终同网格直接抽查待做。',
        '下一段只对两项各最多做两次单变量探针和一次候选，共最多6个额外几何；探针已通过则立即停调。',
        '整体上限仍为30个几何、每配置4个，不能开启第二周期或宽度优化。','',
        '|配置|P|R|折臂mm|网格|基础通过|S21带宽GHz|较高内峰dB|中心联合20带宽GHz|',
        '|---|---:|---:|---:|---:|---|---:|---:|---:|']
    for s in summary:
        t=s['target20'];lines.append(f"|{s['case']}|{s['P']}|{s['R']}|{s['a_mm']}|{s['actual_elements']}|{s['baseline_passed']}|{s['transmission_BW_GHz']:.6f}|{t['worse_inner_peak_dB']:.6f}|{t['joint20_center_band_GHz'][2]:.6f}|")
    (pc.HERE/'INITIAL18_RESULTS.md').write_text('\n'.join(lines)+'\n',encoding='utf-8')
    for file in root.glob('*.py'):ast.parse(file.read_text(encoding='utf-8'),filename=str(file))
    print(dict(initial_completed=18,baseline_passed=16,peak20_positive=6,tuning_segment=str(root),started=False))

    own=[t for t in ledger['tasks'] if t['case']==case];n=len(own)+1

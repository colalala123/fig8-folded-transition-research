import csv,json,tempfile,unittest
from pathlib import Path
import numpy as np
import phase_control as pc
from phase_final_direct import frequency_plan,compare,crossing
from phase_direct_context import reviewed_source
from preserve_discrete_ranges import restore_properties

class DirectPlanTests(unittest.TestCase):
    @classmethod
    def setUpClass(cls):cls.rows=pc.read(pc.HERE/'selected_sources.json')['cases']
    def test_all_sources_reviewed_and_frozen(self):
        self.assertEqual(len(self.rows),18)
        for row in self.rows:reviewed_source(row['source'],row['case'])
    def test_caps_and_frozen_frequency_plans(self):
        for row in self.rows:
            source=Path(row['source']);plan=frequency_plan(pc.read(source/'response_metrics.json'),pc.read(source/'target20_metrics.json'))
            self.assertEqual(plan,row['frequency_review_plan'])
            self.assertLessEqual(len(plan['frequencies_GHz']),24)
            self.assertLessEqual(len(plan['chunks']),6)
            self.assertTrue(all(1<=len(c)<=4 for c in plan['chunks']))
            self.assertEqual(sorted(f for c in plan['chunks'] for f in c),plan['frequencies_GHz'])
    def sample_subset(self,row,folder):
        with (Path(row['source'])/'sparams.csv').open(encoding='utf-8-sig',newline='') as f:
            reader=csv.DictReader(f);fields=reader.fieldnames;rows=list(reader)
        wanted=set(row['requested_frequencies_GHz']);selected=[r for r in rows if round(float(r['Freq_GHz']),2) in wanted]
        self.assertEqual(len(selected),len(wanted))
        with (folder/'sparams.csv').open('w',encoding='utf-8',newline='') as f:
            w=csv.DictWriter(f,fieldnames=fields);w.writeheader();w.writerows(selected)
        return fields,selected
    def test_identical_sample_regression_all18(self):
        # Unit test only: these are interpolated source subsets, NOT HFSS direct
        # data, never exported as study results or used as acceptance evidence.
        for row in self.rows:
            with tempfile.TemporaryDirectory(prefix='unit_direct_',dir=pc.HERE) as d:
                folder=Path(d);self.sample_subset(row,folder);result=compare(row['source'],folder)
                self.assertEqual(result['max_complex_error'],0)
                self.assertTrue(result['numerical_review_passed'],(row['case'],result))
                self.assertTrue(result['sampled_design_supported'],row['case'])
                self.assertEqual(result['both_sampled_inner_peaks_le_minus20'],row['both_inner_peaks_le_minus20'])
    def test_large_error_rejected(self):
        row=self.rows[0]
        with tempfile.TemporaryDirectory(prefix='unit_direct_',dir=pc.HERE) as d:
            folder=Path(d);fields,rows=self.sample_subset(row,folder)
            z=complex(float(rows[0]['S11_real']),float(rows[0]['S11_imag']))+.01
            rows[0]['S11_real']=z.real;rows[0]['S11_imag']=z.imag;rows[0]['S11_dB']=20*np.log10(abs(z))
            with (folder/'sparams.csv').open('w',encoding='utf-8',newline='') as f:
                w=csv.DictWriter(f,fieldnames=fields);w.writeheader();w.writerows(rows)
            self.assertFalse(compare(row['source'],folder)['numerical_review_passed'])
    def test_no_crossing_returns_none_and_zero_endpoints(self):
        self.assertEqual(crossing([1,2],[1,2]),[])
        self.assertEqual(crossing([1,2],[0,1]),[1])
        self.assertEqual(crossing([1,2],[-1,0]),[2])
    def test_discrete_singlepoints_preserved(self):
        class Sweep:pass
        s=Sweep();s.props={};s.auto_update=True
        restore_properties(s,[3.1,3.2,5,9]);self.assertEqual(s.props['RangeType'],'SinglePoints')
        self.assertEqual(len(s.props['SweepRanges']['Subrange']),3)
        self.assertEqual(s.props['Type'],'Discrete')

if __name__=='__main__':unittest.main()

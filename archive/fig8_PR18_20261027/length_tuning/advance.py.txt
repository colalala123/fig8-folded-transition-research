"""Prepare and dispatch one finite-difference step, or stop early. No blind retry."""
import argparse,subprocess,sys
from pathlib import Path
import phase_control as pc
from phase_geometry_context import STEPS,job_name,validate_geometry_budget
from phase_geometry_worker import dc,geometry_gate,design_verdict
from tuning_proposal import propose_probes,propose_candidate
from pilot_queue import error,normalized_error

def reviewed(run):
    spec=pc.read(run/'input.json');m=pc.read(run/'response_metrics.json');ack=pc.read(run/'receiver_ack.json')
    event=pc.read(run/'completion_event.json')
    assert event['status']=='completed' and event['verified_success'] and event['target_thread_id']==pc.TARGET
    assert ack['operation_success'] and ack['visual_review_completed'] and ack['artifacts_verified']
    assert ack['event_sha256']==pc.sha(run/'completion_event.json') and ack['verification_sha256']==pc.sha(run/'receiver_verification.json')
    for x in event['artifacts']:assert pc.sha(run/x['path'])==x['sha256']
    for f,h in spec['parent_evidence_hashes'].items():assert pc.sha(f)==h
    assert m['curve_sha256']==pc.sha(run/'sparams.csv') and m['prominent_valley_count']==3
    return spec,m,{str(run/f):pc.sha(run/f) for f in ['input.json','response_metrics.json','receiver_ack.json','completion_event.json','receiver_verification.json','sparams.csv','state.json']}

def advance(case,solve=False):
    auth=pc.read(pc.HERE/'authorization.json');assert auth['execution_authorized'] and case in pc.ORDER
    ledger=pc.read(pc.HERE/'budget_ledger.json');assert all(t['status']=='reviewed' for t in ledger['tasks'])
    assert not dc.solvers() and not (pc.HERE/'worker.lock').exists() and not (pc.PARENT/'worker.lock').exists()
    assert not (Path('J:/f8_261007')/'protocol.lock').exists()
    for c in pc.ORDER[:pc.ORDER.index(case)]:assert (pc.HERE/'cases'/c/'selection.json').exists()
    selected=pc.HERE/'cases'/case/'selection.json'
    if selected.exists():print(pc.read(selected));return
    initial=pc.BASE/case/'g01_initial';base,m,parents=reviewed(initial)
    parents.update(base['parent_evidence_hashes'])
    assert not design_verdict(m)
    own=[t for t in ledger['tasks'] if t['case']==case];n=len(own)+1
    measurements=[(initial,base,m)]
    for t in own:
        run=Path(t['path']).parent;s,mm,hh=reviewed(run);parents.update(hh);measurements.append((run,s,mm))
    passing=[x for x in measurements if design_verdict(x[2])]
    if passing or n>3:
        best=passing[0] if passing else min(measurements,key=lambda x:normalized_error(error(x[2],6.72)))
        result=dict(case=case,selected_run=str(best[0]),baseline_passed=bool(passing),
            extra_used=len(own),cumulative_geometry=1+len(own),reason='passing_probe_or_candidate_stop' if passing else 'finite_cycle_exhausted',
            final_direct_pending=True,evidence_hashes=parents)
        if solve:pc.write(selected,result)
        print(result);return
    assert len(ledger['tasks'])<6 and 18+len(ledger['tasks'])+1<=30
    prop=propose_probes(base['geometry'],base['geometry'],m,0);assert prop['action']=='probe'
    if n<=2:proposal=prop['probes'][n-1]
    else:
        assert len(measurements)==3
        proposal=propose_candidate(base['geometry'],base['geometry'],m,[x[2] for x in measurements[1:]],
            [x['step_mm'] for x in prop['probes']],2)
        assert proposal['action']=='candidate',proposal
    g=proposal['geometry'];assert geometry_gate(g,base['geometry'])['passed']
    assert all(g[k]==base['geometry'][k] for k in ['Wm','Ws','Ls2','Ls3'])
    run=pc.BASE/case/job_name(n);assert not run.exists(),'Existing attempt: inspect, never re-dispatch'
    if not solve:print(dict(case=case,ordinal=n,proposal=proposal,started=False));return
    for f,h in pc.read(pc.HERE/'parent_evidence.json')['hashes'].items():assert pc.sha(f)==h
    for file in pc.HERE.glob('*.py'):parents[str(file)]=pc.sha(file)
    for file in ['authorization.json','execution_plan.json','validation.json','implementation_provenance.json','hook_live_read_preflight.json','plan.json','parent_evidence.json','CONTINUATION.md','policy.json','mesh_policy_addendum_v2.json','reference_validation.json','hook/config.json']:
        parents[str(pc.HERE/file)]=pc.sha(pc.HERE/file)
    for file in ['tuning_proposal.py','mesh_budget.py']:
        parents[str(pc.ORIGINAL/file)]=pc.sha(pc.ORIGINAL/file)
    pc.write(run/'budget_before.json',ledger);parents[str(run/'budget_before.json')]=pc.sha(run/'budget_before.json')
    pc.write(run/'proposal.json',proposal);parents[str(run/'proposal.json')]=pc.sha(run/'proposal.json')
    spec=dict(base,role=STEPS[n-1],geometry=g,run_dir=str(run),input_path=str(run/'input.json'),
        project_name=f'{case}_PR18_g{n+1:02d}',new_phase_geometry_ordinal=n,historical_geometry_attempts=1,
        phase_solve_ordinal_including_history=1+n,worker_sha256=pc.sha(pc.HERE/'phase_geometry_worker.py'),
        parent_evidence_hashes=parents,created_at=pc.now(),scope='PR18 bounded length-only local cycle')
    pc.write(run/'input.json',spec)
    ledger['new_geometry_reserved']+=1
    ledger['tasks'].append(dict(case=case,role=spec['role'],path=spec['input_path'],input_sha256=pc.sha(run/'input.json'),new_phase_geometry_ordinal=n,status='reserved'))
    pc.write(pc.HERE/'budget_ledger.json',ledger)
    validate_geometry_budget(spec,run/'input.json')
    release=dict(case=case,input_path=spec['input_path'],input_sha256=pc.sha(run/'input.json'),bulk_dispatch_released=False,at=pc.now())
    pc.write(pc.HERE/'supervised_release.json',release)
    # Handoff old completion-only monitor to the actual successor; never reset its timer.
    pc.write(pc.PARENT/'supervised_release.json',release)
    with (pc.HERE/f'{case}_{n}.stdout.log').open('ab') as out,(pc.HERE/f'{case}_{n}.stderr.log').open('ab') as err:
        proc=subprocess.Popen([sys.executable,str(pc.HERE/'phase_geometry_worker.py'),'--spec',spec['input_path']],
            cwd=str(pc.HERE),stdin=subprocess.DEVNULL,stdout=out,stderr=err,creationflags=subprocess.CREATE_NO_WINDOW)
    print(dict(case=case,run=str(run),worker_pid=proc.pid,actual_solver_started=False))

if __name__=='__main__':
    p=argparse.ArgumentParser();p.add_argument('--case',choices=pc.ORDER,required=True);p.add_argument('--solve',action='store_true')
    a=p.parse_args();advance(a.case,a.solve)

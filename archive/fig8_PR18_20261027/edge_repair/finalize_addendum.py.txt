"""Close only reviewed finite addendum; preserve prior failures and source curves."""
from pathlib import Path
import phase_control as pc
from solver_runtime import dc

def main():
    assert not dc.solvers(),'Do not close while solver active'
    ledger=pc.read(pc.HERE/'case_work_ledger.json')
    assert len(ledger['tasks'])==2 and all(x['status']=='reviewed' for x in ledger['tasks'])
    assert ledger['analyze_launches_reserved']==6
    for f,h in pc.read(pc.HERE/'implementation_hashes.json').items():assert pc.sha(f)==h,f
    results=[]
    for case in pc.ORDER:
        run=pc.BASE/case/'edge_repair';a=pc.read(run/'receiver_ack.json');v=pc.read(run/'receiver_verification.json');s=pc.read(run/'state.json');p=pc.read(run/'input.json')
        assert a['visual_review_completed'] and a['operation_success']
        assert a['event_sha256']==pc.sha(run/'completion_event.json') and a['verification_sha256']==pc.sha(run/'receiver_verification.json')
        assert a['scoped_accepted'] and a['numerical_review_passed']
        assert s['same_mesh_verified'] and s['source_unchanged'] and len(s['segments'])==3 and len(p['requested_frequencies_GHz'])==12
        assert all(x['observed_16core_process'] and not x['observed_adaptive_process'] for x in s['segments'])
        for f,h in v['artifacts_verified'].items():assert pc.sha(f)==h,f
        c=pc.read(run/'comparison.json');assert c==v['comparison'] and c['repaired_direct_labels_accepted']
        results.append(dict(case=case,actual_elements=s['actual_elements'],new_native_points=12,Analyze_launches=3,comparison=c,receiver_ack_sha256=pc.sha(run/'receiver_ack.json'),event_sha256=pc.sha(run/'completion_event.json')))
    out=dict(at=pc.now(),stage='PR18_edge_repair',bounded_execution_complete=True,all_corrected_local_edge_labels_accepted=True,
             original18_interpolating_sweep_consistency_passed=16,corrected_direct_label_addenda=2,
             original_failures_preserved=True,full_matrix_approved=False,new_geometry=0,new_adaptive_meshes=0,new_native_points=24,new_Analyze_launches=6,
             accumulated_final_review_points=382,accumulated_final_review_Analyze=100,
             source_curve_files_modified=False,fullcurve_proxy_training_rows_still=50,local_PR_formula_unchanged=True,
             remaining_original_curve_caveat='P04G05/P05G05 original interpolating curves are not re-certified. Use corrected direct edge labels; do not relabel old failed review as passed.',cases=results)
    pc.write(pc.HERE/'corrected_labels.json',out)
    pc.write(pc.HERE/'closure.json',dict(at=pc.now(),bounded_execution_complete=True,reviewed=2,repaired_direct_labels_accepted=2,new_geometry=0,new_adaptive_meshes=0,new_points=24,Analyze_launches=6,original_interpolating_sweeps_not_recertified=True))
    last=pc.BASE/pc.ORDER[-1]/'edge_repair'
    pc.write(last/'continuation_resolution.json',dict(event_sha256=pc.sha(last/'completion_event.json'),disposition='batch_complete',reason='Both authorized local edge repairs actually solved, independently checked, visually reviewed and ACKed. Original interpolation caveats retained. No further simulation authorized in this addendum.'))
    text=['# Fig8 两项带边补算结果','',
          '## 结论','',
          '**P04G05、P05G05缺失或不可靠的高频侧−20 dB带边，现均已有夹住阈值的原生离散补算证据，修正标签可用。**',
          '这不是把旧插值结果改判通过，也不是将两项几何调成全带−20 dB达标。原来的插值带边偏差是真实的，旧失败记录和原始曲线全部保留。','',
          '|配置|网格|原插值带边 / GHz|修正离散带边 / GHz|相差 / MHz|粗细嵌套带边差 / MHz|修正高频独立通带宽 / GHz|',
          '|---|---:|---:|---:|---:|---:|---:|']
    for r in results:
        c=r['comparison'];d=c['edge_refinement']
        text.append(f"|{r['case']}|{r['actual_elements']:,}|{c['original_interpolation_edge_GHz']:.9f}|{c['corrected_edge_GHz']:.9f}|{c['corrected_vs_original_edge_shift_MHz']:.3f}|{d['coarse_fine_shift_GHz']*1000:.4f}|{c['corrected_high_band_width_GHz']:.9f}|")
    text+=['','## 做了什么','',
           '- 16核、单并行组，P04审查后才启动P05；每项12个新频点、3次Analyze，每次4点，共24点/6次。',
           '- 两项几何、端口、材料、原生基模参考和自适应网格均未改变；每批解前解后网格历史逐项一致。',
           '- 仅修正高频独立通带下边界。所有其他指标继承上一轮已核验的有限离散证据。',
           '- 实际夹住−20 dB阈值，细夹区≤10 MHz；使用相邻更外层的粗夹区核对，带边/由其决定的带宽差仍须≤20 MHz。没有外推缺失交点，没有放宽20 MHz标准。',
           '- 粗、细夹区共享本次离散采样，不是独立网格收敛验证；夹区内线性插值仅用于报告交点，导出的原生仿真数据未混入合成频点。',
           '- 两项叠图均实际打开审查，完成hook已送达当前聊天；结束后关闭本次完成心跳。','',
           '## 对研究结论和模型的影响','',
           '- 固定P/R规律、S21带宽公式、中央联合−20 dB带宽不变；修正的是另一个断开的高频通带边界，不能跨越不合格内峰拼成一个带宽。',
           '- 两项仍是“双内峰没有同时≤−20 dB”的科学负例，不可写成全带−20 dB设计通过。',
           '- 当前账目应写为：**16项原插值有限一致性通过 + 2项修正离散带边标签完成**，不是“18项原插值曲线全部通过”。',
           '- 未重训或擅改已有代理模型。全曲线代理仍用50行，仍排除这两条未重新认证的原插值曲线；各项可用的直接指标与本修正标签分层保存。',
           '- 无需为消除这两个待查带边继续追加仿真；如果要把两条原插值全曲线也重新认证，应另订有限插值精度复核方案，不能冒称本次已完成。','',
           '## 修正的联合−20 dB区间','']
    for r in results:
        text.append(f"- {r['case']}: "+'；'.join(f'[{lo:.9f}, {hi:.9f}] GHz' for lo,hi in r['comparison']['corrected_joint20_intervals_GHz']))
    text+=['','所有区间同时要求S11≤−20 dB、S21≥−2 dB；仅为有限原网格证据，不是全频极值或完整多模网络验收。','',
           f"机器可读证据：`{pc.HERE/'corrected_labels.json'}`。原阶段闭账文件未重开。",'']
    report=pc.HERE/'REPORT.md';report.write_text('\n'.join(text),encoding='utf-8')
    delivery=pc.PARENT.parent/'fig8_PR18_delivery_20261027'
    (delivery/'EDGE_REPAIR_ADDENDUM.md').write_text('\n'.join(text),encoding='utf-8')
    active=dict(at=pc.now(),stage='edge_repair_addendum_complete',operation='completed_with_corrected_direct_edge_labels',solver_running=False,report=str(report),old_interpolation_caveat_retained=True)
    for folder in [pc.HERE,pc.PARENT,pc.PARENT/'length_tuning']:
        pc.write(folder/'active.json',active)
    evidence={str(f):pc.sha(f) for f in pc.HERE.rglob('*') if f.is_file() and f.suffix in ('.py','.md','.png','.json') and 'hook' not in f.parts and '__pycache__' not in f.parts and f.name!='manifest.json'}
    evidence[str(delivery/'EDGE_REPAIR_ADDENDUM.md')]=pc.sha(delivery/'EDGE_REPAIR_ADDENDUM.md')
    pc.write(pc.HERE/'manifest.json',dict(files=evidence,original_scientific_implementation_hashes_verified=1194))
    print(dict(report=str(report),accepted_corrected_labels=2,new_points=24,new_Analyze=6,old_interpolation_caveat_retained=True))
if __name__=='__main__':main()

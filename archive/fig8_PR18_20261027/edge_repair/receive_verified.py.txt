"""Independent receiver check for candidate_direct events; never auto-accepts."""
import argparse
from pathlib import Path
import numpy as np
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import recovery_control  # Establish parent runtime path before importing it.
from phase_final_direct import compare, passes
from case_reference import verify_reference
from recovery_control import HERE,read,write,sha,now
from phase_geometry_publish import hook


def review(run):
    run=Path(run).resolve();event=read(run/'completion_event.json')
    assert event['status']=='completed' and event['verified_success']
    verified={}
    for a in event['artifacts']:
        p=(run/a['path']).resolve();assert p.is_relative_to(run) and sha(p)==a['sha256'];verified[str(p)]=a['sha256']
    plan=read(run/'input.json');state=read(run/'state.json');source=Path(plan['source'])
    assert event['target_thread_id']=='REDACTED-ID'
    assert 1<=len(plan['requested_frequencies_GHz'])<=24 and len(plan['chunks'])<=6
    for p,digest in plan['input_parent_hashes'].items():assert sha(p)==digest
    reference=verify_reference(run)
    if 'reference_treatment' in state:assert reference==state['reference_treatment']
    else:
        # The completed small-batch worker wrote and hashed the authoritative
        # reference_review.json, but omitted its redundant copy in state.
        # Validate the original native matrix/reference file directly; never
        # edit the event-bound state or manufacture an inherited field.
        assert reference==read(run/'reference_review.json')
        assert reference['treatment']=='native_fundamental' and reference['no_conversion']
    assert state['status']=='completed' and state['same_mesh_verified'] and state['source_unchanged'] and state['copied_original_curves_verified']
    history=passes(source/'convergence.txt')
    assert history==passes(run/'convergence_before.txt')==passes(run/'convergence_after.txt')
    assert history[-1][1]==state['actual_elements']
    assert len(state['segments'])==len(plan['chunks']) and all(s['observed_16core_process'] and not s['observed_adaptive_process'] for s in state['segments'])
    for name in ['sparams.csv','all_modes_sparams.csv','mode_impedance_gamma.csv']:
        data=np.genfromtxt(run/name,delimiter=',',skip_header=1)
        assert len(data)==len(plan['requested_frequencies_GHz']) and np.isfinite(data).all()
        assert np.allclose(data[:,0],plan['requested_frequencies_GHz'])
    from merge_native import partial_rows,exported_rows,add_disjoint
    merged=read(run/'merge_sources.json');assert merged['no_interpolation'] and merged['all_chunk_provenance_preserved']
    rows={}
    for origin in merged['sources']:
        for file,digest in origin['hashes'].items():
            assert sha(file)==digest,file
            if 'convergence' in Path(file).name:assert passes(Path(file))==history
        if origin['kind']=='preserved_successful_prefix':
            path=next(Path(f) for f in origin['hashes'] if Path(f).name=='diagnosis_partial_arrays.json')
            actual=partial_rows(read(path));assert len(actual)==6
        else:
            folder=next(Path(f).parent for f in origin['hashes'] if Path(f).name=='all_modes_sparams.csv')
            actual=exported_rows(folder)
        assert sorted(actual)==origin['frequencies'];add_disjoint(rows,actual)
    assert sorted(rows)==plan['requested_frequencies_GHz'] and rows==exported_rows(run)
    result=compare(source,run);assert result==read(run/'comparison.json')
    original=np.genfromtxt(source/'sparams.csv',delimiter=',',names=True)
    direct=np.genfromtxt(run/'sparams.csv',delimiter=',',names=True)
    fig,ax=plt.subplots(figsize=(9,4))
    ax.plot(original['Freq_GHz'],original['S11_dB'],label='Original interpolating sweep')
    ax.scatter(direct['Freq_GHz'],direct['S11_dB'],label='New native discrete solves',color='tab:orange')
    old=np.genfromtxt(Path(result['prior_direct_run'])/'sparams.csv',delimiter=',',names=True)
    win=(old['Freq_GHz']>=direct['Freq_GHz'].min()-.02)&(old['Freq_GHz']<=direct['Freq_GHz'].max()+.02)
    ax.scatter(old['Freq_GHz'][win],old['S11_dB'][win],marker='x',label='Prior native discrete')
    ax.axhline(-20,color='black',linestyle='--');ax.set(xlim=(direct['Freq_GHz'].min()-.015,direct['Freq_GHz'].max()+.015),ylim=(-20.4,-19.6),xlabel='GHz',ylabel='S11 dB',title=plan['case']+' unchanged-mesh edge correction')
    ax.grid(alpha=.2);ax.legend();fig.tight_layout()
    plot=HERE/'cases'/plan['case']/'edge_repair_overlay.png';plot.parent.mkdir(parents=True,exist_ok=True);fig.savefig(plot,dpi=150);plt.close(fig)
    out=dict(event_id=hook.event_id(event),event_sha256=sha(run/'completion_event.json'),at=now(),artifacts_verified=verified,same_mesh_verified=True,actual_elements=state['actual_elements'],observed_16core_process=all(s['observed_16core_process'] for s in state['segments']),comparison=result,plot=str(plot),plot_sha256=sha(plot),visual_review_pending=True,full_matrix_approved=False)
    write(run/'receiver_verification.json',out)
    print(dict(plot=str(plot),numerical_review_passed=result['numerical_review_passed'],corrected_edge_GHz=result['corrected_edge_GHz'],original_interpolation_consistency_passed=result['original_interpolation_consistency_passed']))

if __name__=='__main__':
    p=argparse.ArgumentParser();p.add_argument('--run',type=Path,required=True);review(p.parse_args().run)

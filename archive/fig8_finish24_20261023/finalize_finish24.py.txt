OUT = HERE/'deliverables'
TERMINAL = {'preserved_accepted', 'scoped_accepted', 'preserved_design_failed',
            'bounded_design_failed', 'unresolved_numerical'}
ACCEPTED = {'preserved_accepted', 'scoped_accepted'}
LABELS = {'preserved_accepted':'历史通过（保留）', 'scoped_accepted':'基模范围通过',
          'preserved_design_failed':'历史设计未过', 'bounded_design_failed':'有界设计未过',
          'unresolved_numerical':'数值未解决'}


def collect():
    plan = read(HERE/'plan.json')
    assert len(plan['cases']) == 24
    rows, evidence, curves = [], {}, []
    refs = 0
    for item in plan['cases']:
        case = item['case']
        source = HERE/'cases'/case/'result.json'
        if not source.exists():
            source = Path(item['legacy_result'])
            assert sha(source) == item['legacy_result_sha256']
        result = read(source)
        state = result.get('case_state', item['state'])
        assert state in TERMINAL, (case, state)
        evidence[str(source)] = sha(source)
        for path, digest in result['evidence_hashes'].items():
            assert sha(path) == digest, path
            evidence[path] = digest
            refs += 1
        run = Path(result['selected_run'])
        metrics = result['metrics']
        curve = run/'sparams.csv'
        assert sha(curve) == metrics['curve_sha256']
        evidence[str(curve)] = sha(curve)
        arr = np.genfromtxt(curve, delimiter=',', names=True)
        freq = arr['Freq_GHz']
        def db(prefix):
            if prefix+'_dB' in arr.dtype.names:
                return arr[prefix+'_dB']
            return 20*np.log10(np.maximum(arr[prefix+'_mag'], 1e-300))
        s11, s21 = db('S11'), db('S21')
        assert len(freq) == 1500 and np.isfinite(np.c_[freq, s11, s21]).all()
        assert np.allclose(freq, np.arange(1,1501)/100)
        history = run/'convergence.txt'
        if not history.exists() and run.name == 'verified_recovery':
            # The only derived export directory: bind its unchanged parent solve.
            assert read(run/'normalization_recovery_review.json')['passed']
            assert read(run.parent/'state.json')['geometry'] == result['geometry']
            history = run.parent/'convergence.txt'
        elements = result.get('actual_elements')
        if history.exists():
            elements = passes(history)[-1][1]
            evidence[str(history)] = sha(history)
            if result.get('actual_elements'):
                assert elements == result['actual_elements']
        assert elements is not None
        reference_review = run/'normalization_recovery_review.json'
        reference = dict(treatment='selected source export retained; no new normalization',
            normalization_recovery_review=str(reference_review) if reference_review.exists() else None,
            note='If normalization recovery exists, preserve that calibrated reference transformation; do not relabel every historical curve native or assume a universal50ohm reference.')
        if reference_review.exists():
            evidence[str(reference_review)] = sha(reference_review)
        comparison = result.get('comparison', result.get('direct_comparison', {}))
        direct = {
            'BW_GHz': comparison.get('direct_BW_GHz', comparison.get('direct_bracket_BW_GHz')),
            'peak_difference_dB': comparison.get('direct_peak_difference_dB'),
            'central_valley_GHz': comparison.get('direct_central_valley', {}).get('GHz'),
            'max_complex_error': comparison.get('max_complex_error', comparison.get('max_fundamental_complex_error')),
            'BW_shift_GHz': comparison.get('BW_shift_GHz'),
            'points': comparison.get('points'),
            'run': result.get('direct_run'),
        }
        threshold = HERE/'cases'/case/'threshold_review.json'
        threshold_data = read(threshold) if threshold.exists() else None
        if threshold_data:
            evidence[str(threshold)] = sha(threshold)
            direct['peak_difference_dB'] = threshold_data['direct_sample_peak_difference_dB']
            if 'direct_sample_central_valley_GHz' in threshold_data:
                direct['central_valley_GHz'] = threshold_data['direct_sample_central_valley_GHz']
        if state == 'scoped_accepted':
            assert result['design_passed'] and result['numerical_review_passed']
        ledger = result.get('geometry_budget') or attempt_ledger(case)
        assert ledger['attempts_used_or_reserved'] <= 8
        row = dict(case=case, state=state, state_label=LABELS[state],
            included_in_accepted_subset=state in ACCEPTED,
            acceptance_tier='historical_preserved' if state=='preserved_accepted' else
                ('fundamental_same_mesh_direct' if state=='scoped_accepted' else 'not_accepted'),
            geometry_mm=result['geometry'], selected_run=str(run), result_path=str(source),
            source_reference=reference,
            result_sha256=sha(source), selected_curve_sha256=sha(curve), actual_elements=elements,
            geometry_attempts_including_history=ledger['attempts_used_or_reserved'],
            selected_curve_metrics=metrics, direct_review=direct,
            design_passed=result.get('design_passed', state=='preserved_accepted'),
            numerical_review_passed=result.get('numerical_review_passed'),
            numerical_status=result.get('numerical_status', 'historical_policy_preserved' if state.startswith('preserved') else state),
            reason=result['reason'], risk_note=result.get('risk_note', result.get('scope')),
            threshold_warning=threshold_data, full_matrix_approved=False)
        rows.append(row)
        curves.append(dict(case=case, state=state, acceptance_tier=row['acceptance_tier'],
            source_reference=reference,
            source=str(curve), source_sha256=sha(curve), frequency_GHz=freq.tolist(),
            S11_dB=s11.tolist(), S21_dB=s21.tolist()))
    assert len({r['case'] for r in rows}) == 24
    counts = dict(Counter(r['state'] for r in rows))
    assert counts == dict(preserved_design_failed=2, scoped_accepted=14,
        preserved_accepted=4, bounded_design_failed=3, unresolved_numerical=1)
    evidence[str(HERE/'plan.json')] = sha(HERE/'plan.json')
    evidence[str(HERE/'policy.json')] = sha(HERE/'policy.json')
    evidence[str(Path(__file__).resolve())] = sha(__file__)
    return rows, curves, evidence, refs, counts


def paired_summary(rows):
    accepted = {r['case']:r for r in rows if r['included_in_accepted_subset']}
    pairs = []
    for i in range(1, 13):
        a, b = accepted.get(f'D{i:02}G02'), accepted.get(f'D{i:02}G05')
        if a is None or b is None:
            continue
        ba = a['selected_curve_metrics']['passband_2dB']['BW_GHz']
        bb = b['selected_curve_metrics']['passband_2dB']['BW_GHz']
        pairs.append(dict(D=f'D{i:02}', G02=a['case'], G05=b['case'],
            Wm_mm=a['geometry_mm']['Wm'], Ws_mm=a['geometry_mm']['Ws'],
            BW_G02_GHz=ba, BW_G05_GHz=bb, delta_BW_GHz=bb-ba,
            Ls1_G02_mm=a['geometry_mm']['Ls1'], Ls1_G05_mm=b['geometry_mm']['Ls1'],
            Lm_G02_mm=a['geometry_mm']['Lm'], Lm_G05_mm=b['geometry_mm']['Lm'],
            limitation='Retuned geometries, preserved/new numerical policies, and source-reference treatment retained; descriptive association, not an isolated arm-length effect.'))
    return dict(pairs=pairs, n=len(pairs), positive=sum(p['delta_BW_GHz']>0 for p in pairs),
        mean_delta_BW_GHz=float(np.mean([p['delta_BW_GHz'] for p in pairs])),
        scope='Selected interpolated curves only, with accepted tiers retained. No statistical independence or causal law claimed.')


def render_plots(rows, pairs):
    fig, ax = plt.subplots(figsize=(12, 5.6), layout='constrained')
    colors = {'preserved_accepted':'#549bc4','scoped_accepted':'#1a786d',
        'preserved_design_failed':'#b96736','bounded_design_failed':'#bd4f38','unresolved_numerical':'#9b61a8'}
    for state in TERMINAL:
        group = [(i,r) for i,r in enumerate(rows) if r['state']==state]
        ax.scatter([i for i,r in group], [r['selected_curve_metrics']['passband_2dB']['BW_GHz'] for i,r in group],
            s=62, color=colors[state], label=state, zorder=3)
    ax.set_xticks(range(24), [r['case'] for r in rows], rotation=65, ha='right')
    ax.set(ylabel='Selected-curve S21 >= -2 dB bandwidth (GHz)', title='Fig8 first24: final dispositions (not all accepted)')
    ax.grid(axis='y', alpha=.25)
    ax.legend(fontsize=8, ncol=2)
    fig.savefig(OUT/'bandwidth_status.png', dpi=170)
    plt.close(fig)
    fig, ax = plt.subplots(figsize=(9, 4.5), layout='constrained')
    ps = pairs['pairs']
    ax.bar([p['D'] for p in ps], [p['delta_BW_GHz']*1000 for p in ps], color='#337f87')
    ax.axhline(0, color='#555555', lw=.8)
    ax.set(ylabel='BW(G05) - BW(G02), MHz', title='Accepted pairs only: descriptive association, retuning confounded')
    ax.grid(axis='y', alpha=.2)
    fig.savefig(OUT/'accepted_pair_bandwidth.png', dpi=170)
    plt.close(fig)

def main():
    assert not dc.solvers(), 'A solver remains active; do not close batch yet'
    rows, curves, evidence, refs, counts = collect()
    assert not (OUT/'manifest.json').exists(), 'Do not overwrite a frozen delivery'
    OUT.mkdir(exist_ok=True)
    pairs = paired_summary(rows)
    meta = dict(schema='fig8_finish24_delivery_v1', generated_at=now(), full_matrix_approved=False,
        scope='Folded Fig8 first24 only; uniform data excluded; final disposition does not imply all accepted.')
    write(OUT/'all24_results.json', dict(**meta, counts=counts, rows=rows))
    write(OUT/'accepted18_results.json', dict(**meta, rows=[r for r in rows if r['included_in_accepted_subset']]))
    write(OUT/'selected_curves.json', dict(**meta, quantity='selected-source fundamental magnitude only; reference treatment retained per case', curves=curves))
    write(OUT/'paired_bandwidth.json', {**meta, **pairs})
    write(OUT/'evidence_audit.json', dict(**meta, formal_evidence_references_verified=refs,
        unique_sources_verified=len(evidence), mismatches=[], evidence_hashes=evidence))
    render_plots(rows, pairs)
    write_report(rows, pairs, refs, counts)
    files = {str(p.name):sha(p) for p in sorted(OUT.iterdir()) if p.is_file()}
    write(OUT/'manifest.json', dict(**meta, counts=counts, file_sha256=files,
        final_event_path=str(BASE/'D04G02/c01_candidate/completion_event.json'),
        final_event_sha256=sha(BASE/'D04G02/c01_candidate/completion_event.json')))
    print(json.dumps(dict(counts=counts, evidence_references=refs, unique_evidence=len(evidence),
        accepted=18, pairs=pairs, delivery=str(OUT)), ensure_ascii=False))


if __name__ == '__main__':
    main()

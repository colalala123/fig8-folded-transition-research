import unittest
from tuning_proposal import propose_probes,propose_candidate,assess_trial

def metrics(peak,center):
    return {'prominent_valley_count':3,'signed_peak_difference_dB':peak,'peak_difference_dB':abs(peak),
            'central_valley_GHz':center,'validity':dict(S11_below_minus10_between_valleys=True,
             S21_ge_minus2_between_valleys=True,band_not_truncated=True,nonpositive_transmission=True)}

G={'Wm':2.,'Ws':.3,'Lm':6.499465,'Ls1':11.84066,'Ls2':1.15,'Ls3':2.1}

class TestProposals(unittest.TestCase):
    def test_probe_budget_reserved(self):
        self.assertEqual(propose_probes(G,G,metrics(1.,6.86),6)['action'],'stop')
        self.assertEqual(len(propose_probes(G,G,metrics(1.,6.86),1)['probes']),2)
    def test_candidate_step_is_bounded(self):
        r=propose_candidate(G,G,metrics(1.,6.86),[metrics(1.2,6.86),metrics(1.,6.87)],[.1,.05],3)
        self.assertEqual(r['action'],'candidate')
        self.assertLessEqual(abs(r['delta_mm'][0]),.3+1e-10);self.assertLessEqual(abs(r['delta_mm'][1]),.1+1e-10)
    def test_singular_model_stops(self):
        r=propose_candidate(G,G,metrics(1.,6.86),[metrics(1.,6.86),metrics(1.,6.86)],[.1,.05],3)
        self.assertEqual(r['action'],'stop')
    def test_design_pass_does_not_trigger_more_geometry(self):
        self.assertEqual(propose_probes(G,G,metrics(.2,6.72),8)['action'],'direct_review')
    def test_failed_trial_no_unbounded_loop(self):
        self.assertEqual(assess_trial(metrics(1.,6.86),metrics(2.,6.9),8)['action'],'stop')
        self.assertEqual(assess_trial(metrics(1.,6.86),metrics(2.,6.9),4)['action'],'one_half_step')

if __name__=='__main__':unittest.main()

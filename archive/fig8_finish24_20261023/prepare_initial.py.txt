"""Freeze one remaining initial geometry, without launching any solver."""
import argparse
from control import HERE, BASE, OLD, read, write, sha, now
from geometry_worker import geometry_gate, dc
from pathlib import Path

def prepare(case):
    assert case in ['D11G02','D11G05','D12G02'],'Only existing unrun initial cases'
    assert not dc.solvers() and not (HERE/'worker.lock').exists() and not (OLD/'protocol.lock').exists()
    plan=read(HERE/'plan.json');row=next(c for c in plan['cases'] if c['case']==case)
    assert row['action']=='initial_geometry_solve' and not row['selected_source']
    assert row['historical_phase_solves']==0 and row['existing_new_attempts']==0
    previous=read(HERE/'supervised_release.json');previous_input=read(previous['input_path'])
    ack=read(Path(previous_input['run_dir'])/'receiver_ack.json')
    assert ack['artifacts_verified'] and ack['visual_review_completed']
    assert ack['event_sha256']==sha(Path(previous_input['run_dir'])/'completion_event.json')
    assert geometry_gate(row['initial_geometry'],row['initial_geometry'])['passed']
    run=BASE/case/'g00';assert not run.exists(),'Existing initial must be inspected, never overwritten'
    source=Path('F:/hfss_f8_261002/B0/B0.aedt');addon=HERE/'mesh_policy_addendum_v2.json'
    assert read(addon)['user_authorized'] and read(addon)['reviewed_exception_upper']==112000
    spec=run/'input.json'
    write(spec,{'case':case,'role':'initial_geometry','geometry':row['initial_geometry'],'source':str(source),'source_sha256':sha(source),
          'run_dir':str(run),'policy_sha256':sha(HERE/'policy.json'),'mesh_addendum_sha256':sha(addon),
          'phase_solve_ordinal_including_history':1,'created_at':now(),'worker_sha256':sha(HERE/'geometry_worker.py'),
          'controller_sha256':sha(HERE/'mesh_budget.py')})
    archive=HERE/'supervised_releases'/f"{previous['case']}_{Path(previous_input['run_dir']).name}.json"
    assert not archive.exists();write(archive,previous)
    write(HERE/'supervised_release.json',{'case':case,'input_path':str(spec.resolve()),'input_sha256':sha(spec),'bulk_dispatch_released':False,
          'at':now(),'reason':'Previous supervised geometry reviewed; next single initial case within existing 24-case user authorization.'})
    plan['current_stage']='supervise_new_geometry_initials';write(HERE/'plan.json',plan)
    print(spec)

if __name__=='__main__':
    p=argparse.ArgumentParser();p.add_argument('--case',required=True);prepare(p.parse_args().case)

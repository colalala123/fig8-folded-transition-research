"""One remaining full direct task; permit an already-passing probe unchanged."""
from pathlib import Path
from phase_geometry_context import HERE,BASE,OLD,ORIGINAL,read,write,sha,now,status_report

def validate_spec(path):
    p=read(path);auth=read(HERE/'authorization.json');ledger=read(HERE/'budget_ledger.json')
    assert auth['execution_authorized'] and auth['stage']=='A' and not auth['stage_B_C_released']
    assert p['case']=='D09G02' and p['role']=='cached_phase_direct'
    assert p['cached_task_ordinal_for_case']==2 and p['geometry_budget_increment']==0 and not p['adaptive_allowed']
    assert ledger['new_cached_tasks_reserved']==4==auth['cached_task_cap']
    assert ledger['new_geometry_reserved']==1 and ledger['case_cumulative_geometry']['D09G02']==7
    assert p['historical_geometry_attempts']==7 and p['new_geometry_used']==1 and p['lifetime_geometry_attempts']==8
    assert len(p['frequencies_GHz'])<=70 and 10.19 in p['frequencies_GHz'] and p['frequencies_GHz']==sorted(set(p['frequencies_GHz']))
    assert Path(p['source']).resolve()==(BASE/'D09G02/g01_probe_Ls1').resolve()
    assert Path(p['run_dir']).resolve()==(BASE/'D09G02/final_direct').resolve()
    decision=read(HERE/'D09G02_early_direct_decision.json')
    assert decision['action']=='early_direct_review_of_passing_probe' and decision['design_passed']
    assert decision['selected_role']=='probe_Ls1' and Path(decision['selected_run']).resolve()==Path(p['source']).resolve()
    assert sha(HERE/'phase_final_direct.py')==p['worker_sha256']
    rel=read(HERE/'supervised_release.json')
    assert Path(rel['input_path']).resolve()==Path(path).resolve() and rel['input_sha256']==sha(path) and not rel['bulk_dispatch_released']
    tasks=[t for t in ledger['tasks'] if t['case']=='D09G02' and t['role']=='cached_phase_direct']
    assert len(tasks)==1 and Path(tasks[0]['path']).resolve()==Path(path).resolve() and tasks[0]['input_sha256']==sha(path)
    for f,h in p['parent_evidence_hashes'].items():assert sha(f)==h,f
    return p

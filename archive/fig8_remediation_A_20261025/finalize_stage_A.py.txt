"""Audit and publish an additive stage-A delivery; no solver or original edits."""
from collections import Counter
from datetime import datetime,timezone,timedelta
from pathlib import Path
import copy
import shutil
import numpy as np
import phase_control as pc

OUT=pc.HERE/'deliverables'

def main():
    assert not (OUT/'manifest.json').exists(), 'Frozen delivery already exists'
    original=pc.ORIGINAL/'deliverables'
    original_manifest=pc.read(original/'manifest.json')
    for name,digest in original_manifest['file_sha256'].items():assert pc.sha(original/name)==digest
    closure=pc.read(pc.ORIGINAL/'batch_closure.json')
    for r in closure['case_results']:assert pc.sha(r['result_path'])==r['result_sha256']
    base=pc.read(original/'all24_results.json')
    curves=pc.read(original/'selected_curves.json')
    rows=copy.deepcopy(base['rows']); by_case={r['case']:r for r in rows}
    changes=[];evidence={};refs=0
    for case in ['D11G05','D09G02']:
        result_path=pc.HERE/'cases'/case/'result.json';r=pc.read(result_path)
        assert r['case_state']=='scoped_accepted' and r['numerical_review_passed'] and r['design_passed']
        assert not r['full_matrix_approved']
        for file,digest in r['evidence_hashes'].items():
            assert pc.sha(file)==digest,file
            evidence[file]=digest;refs+=1
        evidence[str(result_path)]=pc.sha(result_path)
        old=by_case[case];source=Path(r['selected_run']);m=r['metrics']
        assert pc.sha(source/'sparams.csv')==m['curve_sha256']
        comparison=r['comparison']['reference62'] if case=='D11G05' else r['comparison']
        changes.append(dict(case=case,previous_state=old['state'],new_state='scoped_accepted',
            previous_result=old['result_path'],previous_result_sha256=old['result_sha256'],
            new_result=str(result_path),new_result_sha256=pc.sha(result_path),
            new_geometry=r['new_geometry_attempts'],cached_tasks=r['cached_tasks_used'],
            actual_elements=r['actual_elements'],direct_metrics=r['direct_verified_metrics'],
            comparison=comparison,old_records_preserved=True))
        row=copy.deepcopy(old)
        row.update(state='scoped_accepted',state_label='A阶段修复后基模范围通过',
            included_in_accepted_subset=True,acceptance_tier='stage_A_fundamental_same_mesh_direct',
            geometry_mm=r['geometry'],selected_run=str(source),result_path=str(result_path),
            result_sha256=pc.sha(result_path),selected_curve_sha256=pc.sha(source/'sparams.csv'),
            actual_elements=r['actual_elements'],geometry_attempts_including_history=r.get('lifetime_geometry_attempts',r['historical_geometry_attempts']),
            selected_curve_metrics=m,design_passed=True,numerical_review_passed=True,
            numerical_status='same_mesh_direct_checks_passed',reason=r['reason'],
            risk_note=r.get('risk_note',r.get('scope')),threshold_warning=None,
            original_disposition=dict(state=old['state'],path=old['result_path'],sha256=old['result_sha256']),
            source_reference=dict(treatment='Native fundamental source; no new normalization',normalization_recovery_review=None,
                interpolation_tolerance_percent=.5 if case=='D11G05' else 1.,
                note='D11 geometry unchanged;0.5% interpolation reviewed against unchanged-mesh previous62direct plus new5local points.' if case=='D11G05' else 'Selected passing Ls1probe retains its original role; complete63point direct includes10.19GHz.'),
            direct_review=dict(BW_GHz=comparison['direct_BW_GHz'],peak_difference_dB=comparison['direct_peak_difference_dB'],
                central_valley_GHz=comparison['direct_central_valley']['GHz'],max_complex_error=comparison['max_complex_error'],
                BW_shift_GHz=comparison['BW_shift_GHz'],points=comparison['points'],run=r.get('direct_run',r.get('previous_direct'))))
        by_case[case].clear();by_case[case].update(row)
        data=np.genfromtxt(source/'sparams.csv',delimiter=',',names=True)
        assert len(data)==1500 and np.isfinite(np.array(data.tolist(),dtype=float)).all()
        cr=next(c for c in curves['curves'] if c['case']==case)
        cr.update(state='scoped_accepted',acceptance_tier=row['acceptance_tier'],source=str(source/'sparams.csv'),
            source_sha256=pc.sha(source/'sparams.csv'),source_reference=row['source_reference'],
            frequency_GHz=data['Freq_GHz'].tolist(),S11_dB=data['S11_dB'].tolist(),S21_dB=data['S21_dB'].tolist())
    counts=dict(Counter(r['state'] for r in rows))
    assert counts==dict(preserved_design_failed=2,scoped_accepted=16,preserved_accepted=4,bounded_design_failed=2)
    assert sum(r['included_in_accepted_subset'] for r in rows)==20
    tasks=[pc.BASE/'D11G05/n01_local12',pc.BASE/'D11G05/n02_interp0p5',pc.BASE/'D09G02/n01_missing10p19',pc.BASE/'D09G02/g01_probe_Ls1',pc.BASE/'D09G02/final_direct']
    for task in tasks:
        event=pc.read(task/'completion_event.json');ack=pc.read(task/'receiver_ack.json');receipt=pc.read(task/'notification_receipt.json')
        assert event['status']=='completed' and event['target_thread_id']==pc.TARGET
        assert receipt['state']=='accepted' and ack['event_sha256']==pc.sha(task/'completion_event.json')
        assert ack['artifacts_verified'] and ack['visual_review_completed']
        for f in ['input.json','completion_event.json','receiver_verification.json','receiver_ack.json','notification_receipt.json']:
            evidence[str(task/f)]=pc.sha(task/f)
    ledger=pc.read(pc.HERE/'budget_ledger.json')
    assert ledger['new_geometry_reserved']==1 and ledger['new_cached_tasks_reserved']==4 and len(ledger['tasks'])==5
    assert all(t['status'].startswith('reviewed') for t in ledger['tasks'])
    for f in [pc.HERE/'authorization.json',pc.HERE/'budget_ledger.json',pc.HERE/'D09G02_early_direct_decision.json',pc.ORIGINAL/'batch_closure.json',original/'manifest.json',Path(__file__).resolve()]:evidence[str(f)]=pc.sha(f)
    OUT.mkdir(exist_ok=True)
    meta=dict(schema='fig8_stage_A_delivery_v1',at=pc.now(),scope='Additive effective view of original24folded cases after stageA; uniform data excluded',original_batch_files_unchanged=True,full_matrix_approved=False)
    pc.write(OUT/'effective_all24_results.json',dict(**meta,counts=counts,rows=rows))
    pc.write(OUT/'effective_accepted20_results.json',dict(**meta,rows=[r for r in rows if r['included_in_accepted_subset']]))
    pc.write(OUT/'effective_selected_curves.json',dict(**meta,quantity='Fundamental magnitude only, per-source reference and interpolation policy retained',curves=curves['curves']))
    pc.write(OUT/'changes_and_budget.json',dict(**meta,changes=changes,new_geometry_used=1,new_geometry_cap=3,cached_tasks_used=4,cached_task_cap=4,total_work_items=5,work_item_cap=7,saved_geometry_points=2,B_C_released=False))
    pc.write(OUT/'evidence_audit.json',dict(**meta,formal_new_evidence_references_verified=refs,unique_evidence_sources=len(evidence),evidence_hashes=evidence,all5completion_acknowledgments_verified=True))
    for source,name in [(pc.HERE/'D11G05_n01_local12_review.png','D11_12GHz_diagnosis.png'),(pc.HERE/'D11G05_n02_review.png','D11_interpolation_repair.png'),(pc.HERE/'cases/D09G02/g01_probe_Ls1_response.png','D09_passed_geometry.png'),(pc.HERE/'cases/D09G02/final_direct_overlay.png','D09_direct_review.png')]:shutil.copy2(source,OUT/name)
    stamp=datetime.now(timezone(timedelta(hours=9))).isoformat(timespec='seconds')
    text=f'''# Fig8 新一轮A阶段结果

生成：{stamp}（Asia/Tokyo）。A阶段已结束；B/C未释放，不继续派发。

## 结论

- 两个目标项均已在新阶段范围通过：D11G05由“数值未解决”变为通过，D09G02由“设计未过且旧直接缺点”变为新几何完整关键点复核通过。
- 原24配置有效结果现在为 **20项通过、4项设计未达标、0项数值未解决**。通过包括4项历史保留与16项范围验收；不是24项全部成功，也不是完整多模网络批准。
- 仍未通过：D01G02、D04G02、D09G05、D12G05。本次没有操作这4项。
- 原首批24闭环、失败记录和交付包未覆盖；本目录为明确标注的新阶段补充有效视图。

## 正式通过指标（直接采样口径）

| 配置 | 网格 | 峰差/dB | 中央谷/GHz | BW₂dB/GHz | 最大复数误差 | BW位移/MHz |
|---|---:|---:|---:|---:|---:|---:|
| D11G05 | 101715 | 0.155491285 | 6.73 | 7.197455148 | 0.004499806 | 4.001853 |
| D09G02 | 102346 | 0.278299511 | 6.73 | 7.251915670 | 0.001005278 | 0.461467 |

BW位移为完整插值曲线相对有限直接采样阈值交点的差，不是网格误差或实际加工误差。新插值BW分别7.201457002、7.251454203GHz，JSON分别保留，不能混成同一口径。

## D11G05：没有用几何变化掩盖数值问题

旧1%插值对12GHz的S21复数偏差0.009655397超过原0.005。5点邻域诊断发现旧直接点在12GHz重复偏差仅4.58e-14，证据指向插值，而非直接解不稳定。未改网格、几何、端口、材料或参考处理，只做一次0.5%插值复核。

同101715网格的原完整62点直接数据与新曲线逐点复算，原数值和采样设计条件均通过；新增5点最大复数偏差0.002476647亦通过。62+5有12GHz重叠，不能说有67个不同频点。0.5%任务快速复用缓存，MCP确实请求16核单任务，但未观察到新的hf3d进程；如实保留，没有为补进程证据重跑，也没有继续0.25%/0.1%加严。

## D09G02：一处长度修正即可，不跑满预算

旧10.19GHz缺失点在原101410网格的一次独立诊断中可正常求解，与原插值最大复数差5.33e-7；未反推旧失败根因已确定，原failed事件及61点部分恢复记录保留。

新几何仅将Ls1由12.48066增至12.58066mm，Lm保持6.483603479215181mm；Wm=1.6、Ws=0.58、Ls2=1.71、Ls3=1.98mm不变。第一个Ls1探针已满足原设计，因此选定该几何直接验证，角色仍是probe_Ls1，不伪造联合候选或增加次数。

新102346网格上完整63个关键直接频点全部有限，包含10.19GHz、双峰、中央谷、带边和原锚点。最大近带S21误差0.004022847dB，峰高误差0.000139959/0.016671796dB，原限值均通过。不是把旧101410网格的61+1点拼入新几何验收。

按原方案“成功分支可提前停止，不为用满上限求解”，省去独立Lm探针与联合候选。因此未测新中心完整Jacobian，不声称得到新的Lm灵敏度。

## 预算和资源

本轮实际 **1个新几何+4个缓存任务=5个工作项**，原上限3+4=7，节省2个新几何。历史未清零：D11G05仍7次几何，D09G02原7+本轮1=8次。缓存复核不计为新几何，一个工作项可能有多个HFSS内部阶段，不能当成Analyze次数。

使用可见AEDT/HFSS2024R2、本机MCP、16核单任务串行请求；有实际求解的任务保存hf3d命令/PID/创建时间，同网格复核无新自适应。D11的纯缓存插值没有新进程观察，单列如上。

5个完成事件均由hook通知本聊天，产物哈希核验、实际曲线查看及ACK闭环均已核对。仅终态通知后5分钟无新求解时才检查AI状态；正常仿真不周期唤醒。本次没有新增全局定时任务，也未声称完成锁屏/休眠专项测试。

## 标准与研究边界

最终峰差≤0.5dB、中央谷6.72±0.03GHz、谷间S11<-10dB/S21≥-2dB等原设计标准未改；复数误差≤0.005、近带S21差≤0.05dB、峰高差≤0.1dB、边缘/BW差≤20MHz等原数值标准未改。

仍是原生基模S11/S21的有限频点范围验收，不是完整多模、连续极值证明或制造容差资格。历史保留与新数值策略/参考处理仍分层；D01G05/D10G02临界通过警告保留。D11采用0.5%插值是单项明确修复，不是默默把全数据重新统一了数值策略。

本次增加可靠通过点，不等于已验证弯折带宽物理规律。P_response由峰高反推、R_BW由带宽反解；不能以BW对R_BW再拟合当独立预测证据。均匀数据未混入，六极及扩展批未运行。

## 数据与下一步

- effective_all24_results.json：新阶段补充后的24项完整有效视图，保留被替代旧结论路径/SHA。
- effective_accepted20_results.json：20项通过子集，保留历史/原范围/本轮范围层级。
- effective_selected_curves.json：24条自包含S11/S21幅度曲线，GHz/dB，1500点/条；非复数网络或可级联工程。
- changes_and_budget.json、evidence_audit.json、manifest.json：变化、预算、来源及SHA。
- 4张实际诊断/验收图作为查阅附件；完整工程和更高阶模式原始数据仍在各源路径。

下一步可选择B阶段处理D04G02和D09G05（方案上限6新几何+2缓存）；C阶段是D01G02/D12G05历史失败项。**本轮只授权A，B/C尚未启动，需用户决定。**
'''
    (OUT/'REPORT.md').write_text(text,encoding='utf-8')
    pc.write(OUT/'manifest.json',dict(**meta,effective_counts=counts,accepted_count=20,not_accepted_count=4,
        new_geometry_used=1,cached_tasks_used=4,total_work_items=5,
        files={p.name:pc.sha(p) for p in sorted(OUT.iterdir()) if p.is_file()},
        last_event_sha256=pc.sha(pc.BASE/'D09G02/final_direct/completion_event.json')))
    print(dict(accepted=20,remaining_design_failed=4,new_geometry=1,cached_tasks=4,
        phase_formal_evidence_references_verified=refs,phase_unique_sources=len(evidence),output=str(OUT)))

if __name__=='__main__':main()

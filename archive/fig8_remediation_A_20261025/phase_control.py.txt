def read(p): return json.loads(Path(p).read_text(encoding='utf-8-sig'))
def sha(p): return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def now(): return datetime.now(timezone.utc).isoformat()
def write(p, data):
    p=Path(p); p.parent.mkdir(parents=True, exist_ok=True)
    tmp=p.with_suffix(p.suffix+'.tmp')
    tmp.write_text(json.dumps(data, indent=2, ensure_ascii=False, default=str), encoding='utf-8')
    os.replace(tmp, p)

def status():
    active=read(HERE/'active.json') if (HERE/'active.json').exists() else {}
    write(HERE/'status.json', dict(updated_at=now(), active=active, stage='A',
        scope='D11G05 numerical diagnostics and D09G02 bounded tuning only',
        geometry_cap=3, cached_task_cap=4, stage_B_C_released=False))

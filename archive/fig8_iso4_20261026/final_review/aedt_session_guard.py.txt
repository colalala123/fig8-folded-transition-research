"""Fail if the explicitly attached AEDT dies; never auto-spawn from a failed solve."""
from pathlib import Path
import psutil
import phase_control as pc

def verify():
    identity=pc.read(pc.HERE/'aedt_identity.json')
    p=psutil.Process(identity['pid'])
    assert p.is_running() and abs(p.create_time()-identity['created'])<.01, 'AEDT desktop identity changed; inspect before reconnecting'
    assert p.name().lower()=='ansysedt.exe'
    return identity

def install(dc):
    if getattr(dc.app_for,'_I4_guard',False):return
    original=dc.app_for
    def guarded(path):
        verify()
        assert Path(path).resolve().is_relative_to(pc.BASE.resolve()), 'I4 may only open its own projects'
        return original(path)
    guarded._I4_guard=True
    dc.app_for=guarded

import math

FINAL=pc.HERE/'final_review'
DEST=pc.HERE/'deliverables'

def peak_mean(peaks):
    return 20*math.log10(sum(10**(p/20) for p in peaks)/len(peaks))

def stats(errors):
    return dict(MAE_MHz=sum(abs(e) for e in errors)/len(errors),
        RMSE_MHz=math.sqrt(sum(e*e for e in errors)/len(errors)),max_absolute_MHz=max(abs(e) for e in errors))

def copy(source,target):
    source,target=Path(source),Path(target);target.parent.mkdir(parents=True,exist_ok=True)
    if target.exists():assert pc.sha(target)==pc.sha(source)
    else:shutil.copy2(source,target)

def main():

    assert not (DEST/'results.json').exists(), 'Preserve existing delivery; review before any revision'

    assert initial['new_geometry_reserved']==4 and all(t['design_passed'] for t in initial['tasks'])
    for f,h in pc.read(FINAL/'implementation_hashes.json').items():assert pc.sha(f)==h,f

    old_path=Path(plan['source_validation_result']);assert pc.sha(old_path)==plan['source_validation_sha256']

    for r in old['cases']:

            direct_ledger_sha256=pc.sha(FINAL/'case_work_ledger.json')),full_matrix_approved=False)

        MAE_le_40MHz=result['frozen_prediction_validation']['MAE_MHz']<=40,

    # The first plot is the actual four S11/S21 traces with finite native checks.
    fig,axes=plt.subplots(2,2,figsize=(12,8),layout='constrained')

            ax.plot(a['Freq_GHz'],a[key+'_dB'],label=key,color=color)

        ax.set(xlim=(0,15),ylim=(-85,1),xlabel='GHz',ylabel='dB',title=f"{r['case']}: BW={r['BW_GHz']:.5f} GHz; {r['actual_elements']} elements")
        ax.legend();ax.grid(alpha=.2)
    fig.savefig(DEST/'curves4.png',dpi=160);plt.close(fig)
    fig,axes=plt.subplots(1,3,figsize=(14,4.5),layout='constrained')
    for arm,marker in [(1.4,'o'),(1.8,'s')]:

            ax.plot(x,[r[key]*scale for r in group],marker=marker,label=f'a={arm:g} mm')

        '|配置|Wm/mm|Ws/mm|a/mm|网格|−2dB BW/GHz|预测误差/MHz|联合BW/GHz|直接复核BW差/MHz|',

        'BW公式：BW/GHz=6.804574332+0.798166452 ln(Wm/1.20mm)+0.866177522 ln(Ws/0.44mm)，ln为自然对数，BW按S21≥−2dB定义。', '',

        '与已完成的V02G02/G05组成同折臂三种宽度搭配：']
    for t in trade:
        lines.append(f"- a={t['a_mm']:.1f} mm：直接确认的−2dB带宽跨度{t['direct_BW_spread_MHz']:.3f} MHz；由窄微带/宽槽到宽微带/窄槽，直接S11内峰平均变化{t['direct_endpoint_peak_mean_change_dB']:.3f} dB，平均及较差峰均逐级改善={t['direct_matching_monotonic']}；插值联合带宽变化{t['interpolated_endpoint_joint_BW_change_MHz']:+.3f} MHz。")

        for name in ['plan.json','continuation_plan.json','policy.json','authorization.json','execution_plan.json',
            'budget_ledger.json','case_work_ledger.json','predictions_frozen.json','implementation_provenance.json',

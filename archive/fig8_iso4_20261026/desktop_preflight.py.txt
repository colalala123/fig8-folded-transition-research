"""Inspect/start the visible AEDT desktop once; never run Analyze."""
import asyncio
import shutil
from pathlib import Path
import psutil
import phase_control as pc

async def main():
    running=[p.info for p in psutil.process_iter(['pid','name','cmdline','create_time'])
             if (p.info['name'] or '').lower() in ('ansysedt.exe','hf3d.exe')]
    assert not any(p['name'].lower()=='hf3d.exe' for p in running), 'Existing simulation; inspect first'
    candidates=[p for p in running if p['name'].lower()=='ansysedt.exe']
    if candidates:
        assert len(candidates)==1 and '7990' in ' '.join(candidates[0]['cmdline']), 'Ambiguous existing desktop'
        p=psutil.Process(candidates[0]['pid'])
    else:
        assert not (pc.HERE/'aedt_identity.json').exists(), 'Known desktop disappeared; explicit diagnosis required'
        from ansys.aedt.core import Desktop
        desktop=Desktop(version='2024.2', non_graphical=False, new_desktop=True, port=7990, close_on_exit=False)
        p=psutil.Process(desktop.aedt_process_id)
        desktop.release_desktop(close_projects=False,close_on_exit=False)
    pc.write(pc.HERE/'aedt_identity.json',dict(at=pc.now(),pid=p.pid,created=p.create_time(),
        command=p.cmdline(),visible_requested=True,version='2024.2',port=7990,analyze_requested=False))
    assert shutil.disk_usage(pc.BASE.parent).free>15*1024**3
    # Importing the worker now installs an identity guard around all subsequent app access.
    from phase_geometry_worker import dc
    async with dc.connected_client() as client:
        cat=await client.list_tools()
        names={t.name for t in cat.tools}
        assert {'start_analysis','get_analysis_status'} <= names
        result=await dc.session_call(client,'check_aedt_connection',{'port':7990})
        pc.write(pc.HERE/'mcp_connection_preflight.json',dict(at=pc.now(),connection=result,analyze_requested=False))
    print(dict(desktop_pid=p.pid,visible=True,port=7990,solver_started=False))

if __name__=='__main__':asyncio.run(main())

def read(p):return json.loads(Path(p).read_text(encoding='utf-8-sig'))
def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def write(p,d):Path(p).write_text(json.dumps(d,ensure_ascii=False,indent=2,allow_nan=False)+'\n',encoding='utf-8')
def new(p,d):
    with Path(p).open('x',encoding='utf-8') as f:json.dump(d,f,ensure_ascii=False,indent=2,allow_nan=False);f.write('\n')
def now():return datetime.now(timezone.utc).isoformat()
CACHE={}
def verify(p,h):
    p=Path(p);st=p.stat();k=(str(p),st.st_size,st.st_mtime_ns)
    if k not in CACHE:CACHE[k]=sha(p)
    assert CACHE[k]==h,str(p)
def table(p):
    with Path(p).open(encoding='utf-8-sig',newline='') as f:
        data=list(csv.DictReader(f))
    assert len(data)==1500
    for i,row in enumerate(data):
        values={k:float(v) for k,v in row.items()};assert all(math.isfinite(x) for x in values.values())
        assert abs(values['Freq_GHz']-(i+1)/100)<1e-9
    return data
def role_summary(rows):
    bw=[r['prediction_errors']['S21_BW_MHz'] for r in rows]
    peak=[r['prediction_errors']['worse_inner_peak_dB'] for r in rows]
    rel=[abs(x['prediction_errors']['S21_BW_MHz'])/1000/x['scientific_label']['S21_minus2_BW_GHz']*100 for x in rows]
    return dict(geometries=len(rows),families=len({r['family_id'] for r in rows}),
        primary_passed=sum(r['scientific_label']['primary']['passed'] for r in rows),
        secondary_passed=sum(r['scientific_label']['secondary']['passed'] for r in rows),
        primary_pass_cases=[r['case'] for r in rows if r['scientific_label']['primary']['passed']],
        secondary_pass_cases=[r['case'] for r in rows if r['scientific_label']['secondary']['passed']],
        frozen_forward_S21_BW_MAE_MHz=statistics.mean(map(abs,bw)),
        frozen_forward_S21_BW_RMSE_MHz=math.sqrt(statistics.mean(x*x for x in bw)),
        frozen_forward_S21_BW_max_abs_MHz=max(map(abs,bw)),
        frozen_forward_S21_BW_mean_relative_error_percent=statistics.mean(rel),
        frozen_forward_worse_inner_peak_MAE_dB=statistics.mean(map(abs,peak)),
        frozen_forward_worse_inner_peak_RMSE_dB=math.sqrt(statistics.mean(x*x for x in peak)),
        prediction_source='Unchanged deployment52 Ridge, frozen before any new48HFSS; NOT a model refit on these rows')

"""Read-only resource gate. Never starts/stops HFSS or changes another task."""
import json,sys
from pathlib import Path
import psutil
ROOT=Path(__file__).resolve().parent
def read(p):return json.loads(Path(p).read_text(encoding='utf-8-sig'))
def exact(record):
 try:
  p=psutil.Process(record['pid'])
  return abs(p.create_time()-record['created'])<.01 and p.cmdline()==record['command']
 except psutil.NoSuchProcess:return False
 except psutil.AccessDenied:return None
def check():
 if (ROOT/'PAUSE.json').exists() or (ROOT/'closure.json').exists():return dict(ready=False,state='paused_or_closed')
 wait=read(ROOT/'resource_wait.json');ctx=Path(wait['context'])
 row=next(r for r in read(ROOT/'campaign_frozen.json')['cases'] if r['case']==wait['case']);run=Path(row['run_dir'])
 if (run/'state.json').exists():
  return dict(ready=False,state='already_started_do_not_dispatch',run=str(run),supervised_start_present=(run/'supervised_start.json').exists(),terminal_event_present=(run/'completion_event.json').exists())
 alive=[exact(wait['other_worker']),exact(wait['other_desktop'])]
 if any(x is None for x in alive):return dict(ready=False,state='identity_needs_review')
 if any(alive):return dict(ready=False,state='waiting_other_2core_task',other_worker_live=alive[0],other_desktop_live=alive[1])
 identity=read(ctx/'aedt_identity.json')
 if exact(identity) is not True:return dict(ready=False,state='visible_desktop_identity_needs_review')
 processes=[p.info for p in psutil.process_iter(['pid','name','cmdline']) if (p.info['name'] or '').lower() in ['hf3d.exe','ansysedt.exe']]
 solvers=[p for p in processes if p['name'].lower()=='hf3d.exe']
 desktops=[p for p in processes if p['name'].lower()=='ansysedt.exe' and p['pid']!=identity['pid']]
 if solvers or desktops:return dict(ready=False,state='waiting_other_resources',hf3d=solvers,other_desktops=desktops)
 return dict(ready=True,state='ready_for_prelaunch',context=str(ctx),case=row['case'],run=str(run),original_other_identities_exited=True)
if __name__=='__main__':print(json.dumps(check(),ensure_ascii=False,indent=2))

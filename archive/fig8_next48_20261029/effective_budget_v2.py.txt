"""Versioned recovery cost overlay; original campaign files remain immutable."""
import hashlib,json,shutil
from pathlib import Path
ROOT=Path(__file__).resolve().parent

def read(p):return json.loads(Path(p).read_text(encoding='utf-8-sig'))
def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()

def overlay():
    d=read(ROOT/'recovery1_overlay.json')
    assert d['version']==1 and d['case']=='N02_A18' and d['technical_recovery_attempt']==1
    assert d['original_Analyze']==0 and d['geometry_cap']==48 and d['Analyze_cap']==624
    for p,h in d['immutable_hashes'].items():assert sha(p)==h,p
    s=read(Path(d['original_run'])/'state.json')
    assert not s['segments'] and not s['mesh_stages']
    assert read(Path(d['original_run'])/'completion_event.json')['status']=='failed'
    assert not (Path(d['original_run'])/'receiver_ack.json').exists(),'Never give old failed attempt a successful ACK'
    return d

def effective(row,d=None):
    d=d or overlay()
    return {**row,'context':d['context'],'run_dir':d['effective_run']} if row['case']==d['case'] else dict(row)

def check_segments(state,row):
    assert state['case']==row['case'] and state['geometry']==row['geometry']
    segs=state.get('segments',[])
    assert len(segs)<=13 and sum(bool(x['adaptive_requested']) for x in segs)<=12
    assert sum(not x['adaptive_requested'] for x in segs)<=1
    assert all(x['mcp_request']['cores_per_task']==14 and x['mcp_request']['concurrent_tasks']==1 for x in segs)
    assert state.get('resource_recovery_attempt',0)<=1
    return segs

def validate_campaign(extra_analyze=0,case=None,storage=False):
    assert not (ROOT/'closure.json').exists() and not (ROOT/'PAUSE.json').exists()
    assert isinstance(extra_analyze,int) and extra_analyze in (0,1)
    assert not extra_analyze or case
    d=overlay();plan=read(ROOT/'campaign_frozen.json');rows=plan['cases']
    assert len(rows)==48 and len({r['case'] for r in rows})==48
    counts={};count=0;adaptive={};sweeps={}
    for row in rows:
        paths=[Path(row['run_dir'])]
        if row['case']==d['case']:paths.append(Path(d['effective_run']))
        segs=[];started=False
        for run in paths:
            if not (run/'state.json').exists():continue
            s=read(run/'state.json');started=True
            if run==Path(d['effective_run']):assert s['resource_recovery_attempt']==1
            segs+=check_segments(s,row)
        count+=int(started);counts[row['case']]=len(segs)
        adaptive[row['case']]=sum(bool(x['adaptive_requested']) for x in segs)
        sweeps[row['case']]=sum(not x['adaptive_requested'] for x in segs)
        assert len(segs)<=13 and adaptive[row['case']]<=12 and sweeps[row['case']]<=1
    total=sum(counts.values())
    assert count<=48 and total+extra_analyze<=624
    if case:
        own=next(r for r in rows if r['case']==case)
        assert counts[case]+extra_analyze<=13
        if own['ordinal']>2:
            gate=read(ROOT/'pilot_gate.json');assert gate['passed'] and gate['actual_visual_ACKs']==2
            for p,h in gate['evidence_hashes'].items():assert sha(p)==h,p
    if storage:
        free=shutil.disk_usage(plan['D_base']).free/1024**3
        needed=(48-count)*plan['storage_admission']['per_remaining_geometry_GiB']+19
        assert free>=needed,f'D space: {free:.2f}GiB, reserve {needed:.2f}GiB'
    return dict(geometries_started=count,geometry_cap=48,Analyze_used=total,Analyze_cap=624,
        per_case_Analyze=counts,recovery_Analyze=counts[d['case']],original_failed_Analyze=0,
        cores=14,tasks=1,serial=True,overlay_sha256=sha(ROOT/'recovery1_overlay.json'))

if __name__=='__main__':print(json.dumps(validate_campaign(storage=True),ensure_ascii=False,indent=2))

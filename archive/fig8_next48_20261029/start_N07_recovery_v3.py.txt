"""One exclusive human-approved cached-sweep release. Never retries on missing/uncertain start."""
import ast,json,os,re,subprocess,sys,time
from datetime import datetime,timezone
from pathlib import Path
import psutil
import effective_budget_v3 as b
ROOT=Path(__file__).resolve().parent

def write_new(p,d):
    with Path(p).open('x',encoding='utf-8') as h:json.dump(d,h,ensure_ascii=False,indent=2);h.write('\n')

def native_gate():
    identity=b.read(ROOT/'desktop_epoch2_20261030.json')
    live=[p.info for p in psutil.process_iter(['pid','name','cmdline','create_time']) if (p.info['name'] or '').lower() in ('ansysedt.exe','hf3d.exe')]
    assert len(live)==1 and live[0]['name'].lower()=='ansysedt.exe'
    assert live[0]['pid']==identity['pid'] and abs(live[0]['create_time']-identity['created'])<.01 and live[0]['cmdline']==identity['command']
    return live

def exact(pid,created,command):
    p=psutil.Process(pid);assert p.is_running() and abs(p.create_time()-created)<.01 and p.cmdline()==command

def main():
    m=b.manifest();route=m['routes']['N07_A18'];ctx=Path(route['context']);run=Path(route['effective_run'])
    marker=ROOT/'N07_A18_recovery_release_v3.json'
    assert not marker.exists() and not (ctx/'initial_inputs.json').exists() and not (run/'state.json').exists()
    live=native_gate();budget=b.validate_campaign(case='N07_A18',storage=True);assert budget['N07_A18_total_requests']==7
    cache=b.read(ctx/'cache_reopen_proof.json');assert cache['passed'] and not cache['Analyze_requested']
    assert cache['actual_elements']==102882 and cache['actual_passes']==21 and cache['full_history_identical']
    for rel,h in cache['cache_hashes_after'].items():assert b.sha(run/rel)==h,rel
    assert b.read(ctx/'prepared_reopen_gate.json')['passed']
    assert b.read(ctx/'hook_live_read_preflight.json')['supported_read_thread_verified']
    plan=b.read(ctx/'plan.json');assert [r['case'] for r in plan['cases']]==list(m['routes'])
    frozen=b.read(ROOT/'campaign_frozen.json')
    for row in plan['cases']:
        fc=next(r for r in frozen['cases'] if r['case']==row['case']);proof=b.read(ctx/'cases'/row['case']/'configuration.json')
        for key in ['geometry','role','family_id','target','secondary_target']:assert row[key]==fc[key]
        assert proof['passed'] and not proof['solved'] and proof['geometry']==fc['geometry']
        assert b.sha(row['prepared_project'])==proof['project_sha256']
        assert b.sha(row['original_input'])==m['routes'][row['case']]['original_input_sha256']
    for p in ctx.glob('*.py'):ast.parse(p.read_text(encoding='utf-8'))
    results=[];n=0
    for directory in [ROOT,ctx]:
        cmd=[sys.executable,'-m','unittest','discover','-s',str(directory),'-p','test*.py','-v']
        p=subprocess.run(cmd,capture_output=True,text=True);assert p.returncode==0,p.stderr
        match=re.search(r'Ran (\d+) tests?',p.stderr);assert match;n+=int(match.group(1))
        results.append(dict(command=cmd,exit_code=p.returncode,stdout=p.stdout,stderr=p.stderr))
    assert n>=62
    tests=ctx/'tests_prelaunch_v3.json';write_new(tests,dict(passed=True,total=n,results=results))
    audit=ctx/'prelaunch_audit_v3.json';write_new(audit,dict(at=datetime.now(timezone.utc).isoformat(),passed=True,
        tests_passed=n,tests_sha256=b.sha(tests),Analyze_requested=False,new_adaptive_forbidden=True,
        geometry_unchanged=True,cache_reopen_proof_sha256=b.sha(ctx/'cache_reopen_proof.json'),native_resources=live,
        budget=budget,original_requests_retained=7,new_sweep_cap=1,cumulative_N07_cap=8,
        source_hashes={str(p):b.sha(p) for p in ctx.glob('*.py')}))
    parent=b.read(ctx/'parent_evidence.json')
    for p in [tests,audit,Path(__file__),ROOT/'test_epoch2_v3.py',ROOT/'verify_cached_preflight_v3.py',ctx/'cache_reopen_proof.json',
              ROOT/'audit_prelaunch_v3.py',ROOT/'update_status_v3.py']:
        parent['hashes'][str(p)]=b.sha(p)
    (ctx/'parent_evidence.json').write_text(json.dumps(parent,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
    native_gate()
    write_new(marker,dict(at=datetime.now(timezone.utc).isoformat(),pid=os.getpid(),case='N07_A18',
        technical_recovery_attempt=1,overlay_sha256=b.sha(ROOT/'session_epoch2_overlay_v3.json'),audit_sha256=b.sha(audit)))
    subprocess.run([sys.executable,str(ctx/'finalize_and_authorize.py')],check=True)
    spec=b.read(run/'input.json');assert spec['technical_recovery_attempt']==1 and spec['inherited_successful_segments']==6
    assert spec['resume_checkpoint'] and spec['geometry']==next(r['geometry'] for r in frozen['cases'] if r['case']=='N07_A18')
    native_gate()
    subprocess.run([sys.executable,str(ROOT/'continuation_v3.py'),'launch','--case','N07_A18'],check=True)
    proof=b.read(run/'supervised_start.json');assert proof['actual_14core_single_group_observed'] and proof['inherited_observations_excluded']
    assert len(proof['mcp_requests'])==1
    exact(proof['worker_pid'],proof['worker_created'],proof['worker_command'])
    for attempt in range(20):
        daemon=b.read(run/'lifecycle_observer/daemon.json')
        try:exact(daemon['pid'],daemon['created'],daemon['command']);break
        except psutil.NoSuchProcess:
            if attempt==19:raise
            time.sleep(.1)  # Only reread venv launcher -> real observer identity; never relaunch.
    s=b.read(run/'state.json');assert len(s['mesh_stages'])==6 and len(s['segments'])==7 and not s['segments'][-1]['adaptive_requested']
    assert b.validate_campaign()['N07_A18_total_requests']==8
    write_new(ROOT/'N07_A18_recovery_started_v3.json',dict(at=datetime.now(timezone.utc).isoformat(),actual14core=True,
        inherited_observations_not_used=True,new_sweep_requests=1,new_adaptive_requests=0,total_N07_requests=8,
        start_proof_sha256=b.sha(run/'supervised_start.json'),worker_pid=proof['worker_pid'],observer_pid=daemon['pid'],hook_target=b.read(ctx/'hook/config.json')['default_thread_id']))
    print(dict(actual14core_replacement_sweep_started=True,case='N07_A18',new_adaptive=0,new_sweep=1,total_requests=8))

if __name__=='__main__':main()

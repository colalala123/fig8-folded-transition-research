"""One-shot, zero-Analyze native inspection; never saves/restarts/closes AEDT."""
import sys, json, hashlib
from pathlib import Path
from datetime import datetime, timezone
import psutil

ROOT=Path(__file__).resolve().parent
CTX=ROOT.parent/'fig8_next48_b08_20261029'
RUN=Path('D:/hfss_fig8_next48_20261029/b08/N23_A18/g01_initial')
DEST=ROOT/'N23_A18_cache_inspection2'
def sha(p): return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def read(p): return json.loads(Path(p).read_text(encoding='utf-8-sig'))
def record(d):
    d['at']=datetime.now(timezone.utc).isoformat()
    (DEST/'inspection.json').write_text(json.dumps(d,indent=2,default=str),encoding='utf-8')
    print(json.dumps(d,default=str),flush=True)

def main():
    assert not (ROOT/'PAUSE.json').exists()
    DEST.mkdir(exist_ok=False)
    identity=read(CTX/'aedt_identity.json')
    p=psutil.Process(identity['pid'])
    assert abs(p.create_time()-identity['created'])<.01
    assert p.cmdline()==identity['command']
    native=[x.info for x in psutil.process_iter(['pid','name','cmdline','create_time'])
            if (x.info['name'] or '').lower() in ('ansysedt.exe','hf3d.exe')]
    assert len(native)==1 and native[0]['pid']==identity['pid']
    st=read(RUN/'state.json'); project=Path(st['project'])
    bound={str(RUN/f):sha(RUN/f) for f in ('state.json','worker_error.json','completion_event.json')}
    # Project may be saved by the already-running original sweep; bind failed records only.
    d=dict(case='N23_A18',Analyze_requested=False,phase='before_attach',native=native,
           original_hashes=bound,expected_passes=21,expected_elements=102038)
    record(d)
    sys.path.insert(0,str(CTX))
    import phase_geometry_worker as w
    app=w.dc.app_for(project)
    try:
        assert Path(app.project_file).resolve()==project.resolve()
        d['phase']='attached';d['running']=bool(app.are_there_simulations_running)
        record(d)
        if d['running']: return
        setup=app.get_setup('Setup19'); sw=next(s for s in setup.sweeps if s.name=='Sweep1')
        d['setup_solved']=bool(setup.is_solved); d['sweep_solved']=bool(sw.is_solved)
        d['messages']=list(app.odesktop.GetMessages(app.project_name,app.design_name,0))
        d['setup_props']=w.snapshot(setup.props);d['sweep_props']=w.snapshot(sw.props)
        record(d)
        app.export_convergence('Setup19',output_file=str(DEST/'convergence.txt'))
        d['history_unchanged']=w.passes(DEST/'convergence.txt')==w.passes(RUN/'convergence_stage_05.txt')
        assert d['history_unchanged']
        if d['sweep_solved']:
            w.dc.SWEEP='Sweep1';w.dc.export(app,DEST,w.np.arange(1,1501)/100)
            d['native_1500_row_export_verified']=True
        d['phase']='inspected'
    except BaseException as e:
        d['error']=repr(e);record(d);raise
    finally:
        app.release_desktop(close_projects=False,close_desktop=False)
    d['original_records_unchanged']=all(sha(p)==h for p,h in bound.items())
    record(d);assert d['original_records_unchanged']

if __name__=='__main__': main()

def read(p):return json.loads(Path(p).read_text(encoding='utf-8-sig'))
def write(p,v):Path(p).write_text(json.dumps(v,indent=2,ensure_ascii=False),encoding='utf-8')
def profiles(folder,sweep_only=False):
    paths=list(folder.glob('*.aedtresults/**/*.profile'))
    results=[]
    old=read(folder/'state.json').get('profile_hashes_before',{}) if sweep_only else {}
    for p in paths:
        if sweep_only:
            import hashlib
            if old.get(str(p))==hashlib.sha256(p.read_bytes()).hexdigest():continue
        text=p.read_text(encoding='utf-8',errors='replace');items={}
        for stage in (['Frequency Sweep'] if sweep_only else ['Initial Meshing','Adaptive Meshing','Frequency Sweep']):
            lines=[x for x in text.splitlines() if f"ProfileItem('{stage}'" in x]
            if lines:
                times=re.findall(r'\b(\d{2}:\d{2}:\d{2})\b',lines[-1])
                if times:
                    h,m,s=map(int,times[0].split(':'));items[stage]=h*3600+m*60+s
        if len(items)==(1 if sweep_only else 3):results.append({'path':str(p),'stages_seconds':items,'sum_seconds':sum(items.values())})
    assert len(results)<=1,results
    return results[0] if results else None
def passes(path):
    rows=[]
    for line in path.read_text(encoding='utf-8',errors='replace').splitlines():
        m=re.match(r'\s*(\d+)\|\s*(\d+)\|\s*([\d.eE+-]+)\|',line)
        if m:rows.append([int(m[1]),int(m[2]),float(m[3])])
    return rows

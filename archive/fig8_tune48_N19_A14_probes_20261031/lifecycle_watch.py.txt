"""Quiet external B6 worker-exit observer. NO solver/lock operations.

Normal runs produce no chat messages. A confirmed hard exit without a terminal
event produces one separate failure event; it never rewrites the real run.
"""
import argparse,asyncio,hashlib,json,os,subprocess,sys,time
from pathlib import Path
import psutil
ROOT=Path(__file__).resolve().parent;BASE=Path('${WORKSPACE}\\output\\fig8_tune48_fixedPR_20261031\\data\\N19_A14\\probes')
sys.path.insert(0,str(ROOT.parent/'completion-hook-portable-1.0.0'))
import completion_hook as hook
from phase_heartbeat import mutex
def read(p):return json.loads(Path(p).read_text(encoding='utf-8-sig'))
def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def exact_worker(proof,process=psutil.Process):
    try:
        p=process(proof['worker_pid'])
        return abs(p.create_time()-proof['worker_created'])<.01 and p.cmdline()==proof['worker_command'] and p.is_running()
    except psutil.NoSuchProcess:return False
    except psutil.AccessDenied:return None
def valid_run(run):
    run=run.resolve();assert run.is_relative_to(BASE.resolve()) and len(run.relative_to(BASE).parts)==2
    assert (run/'input.json').exists();return run
async def watch(run):
    run=valid_run(run);folder=run/'lifecycle_observer';folder.mkdir(exist_ok=True)
    me=psutil.Process()
    with mutex(folder):hook.write_json(folder/'daemon.json',dict(pid=me.pid,created=me.create_time(),command=me.cmdline(),registered_at=time.time()))
    missing=0;observed=None
    while True:
        if (folder/'stop.json').exists():
            hook.write_json(folder/'status.json',dict(state='observer_stopped',reason=read(folder/'stop.json').get('reason')));return
        if (run/'completion_event.json').exists():
            hook.write_json(folder/'status.json',dict(state='normal_terminal_event_seen',event_sha256=sha(run/'completion_event.json')));return
        proofpath=run/'supervised_start.json'
        if proofpath.exists():
            proof=read(proofpath);assert proof['input_sha256']==sha(run/'input.json')
            script=Path(proof['worker_command'][1]).resolve();assert script.is_relative_to(ROOT.resolve())
            observed=proof;alive=exact_worker(proof);missing=missing+1 if alive is False else 0
            if missing>=4:break
        # Startup without proof is not falsely classified as a failed solve.
        await asyncio.sleep(5)
    assert observed and exact_worker(observed) is False
    fault=run/'lifecycle_failure';fault.mkdir(exist_ok=True)
    config=hook.load_config(script.parent/'hook/config.json')
    if (run/'completion_event.json').exists():return
    hook.write_json(fault/'worker_evidence.json',dict(worker_proof=observed,input_sha256=sha(run/'input.json'),confirmed_absent_checks=missing,original_state=read(run/'state.json'),
        observed_hf3d=[p.info for p in psutil.process_iter(['pid','name','cmdline']) if (p.info['name'] or '').lower()=='hf3d.exe'],
        cause='Exact worker identity exited without terminal event; cause unknown',solver_failure_not_assumed=True))
    event=fault/'completion_event.json'
    if not event.exists():event=hook.create_event(fault,'failed','长度校正工作进程已硬退出且未生成终态事件；外部只读监护发现，先检查原HFSS和缓存，不重复Analyze。',['worker_evidence.json'],config=config,verified_success=False,purpose='External lifecycle fault only; not a design failure or permission for extra simulation')
    def prompt(e,p):
        return (f'[Fig8新48点worker监护故障] {p}\n工作项：{run}\n'
          f'先读{ROOT/"CONTINUATION.md"}。已确认真实worker身份退出，但不能据此认为HFSS求解失败。'
          '先检查原HFSS/solver及现有缓存，仍运行则等，不清工程锁、不重复Analyze。每项最多一次技术恢复，继承新固定P/R调谐预算（每原项最多4几何、全轮172几何2236Analyze），不重开原48。'
          '原工作项已有终态则接收该事件，不重跑。正常仿真静默，用户暂停优先。')
    hook.prompt_for=prompt
    receipt=await hook.deliver(event,config=config);hook.write_json(fault/'notification_receipt.json',receipt)
    hook.write_json(folder/'status.json',dict(state='fault_notified' if receipt.get('state')=='accepted' else 'delivery_needs_review',fault_event=str(event)))
    if receipt.get('state')=='accepted':
        with (folder/'heartbeat_arm.log').open('ab') as log:
            subprocess.Popen([sys.executable,str(script.parent/'phase_heartbeat.py'),'arm','--event',str(event)],cwd=str(script.parent),stdin=subprocess.DEVNULL,stdout=log,stderr=log,creationflags=subprocess.CREATE_NO_WINDOW)
def _ensure_locked(run):
    run=valid_run(run);folder=run/'lifecycle_observer';folder.mkdir(exist_ok=True);path=folder/'daemon.json'
    if (folder/'stop.json').exists():return dict(disabled=True,reason=read(folder/'stop.json').get('reason'))
    if path.exists():
        r=read(path)
        try:
            p=psutil.Process(r['pid'])
            if abs(p.create_time()-r['created'])<.01 and p.cmdline()==r['command'] and p.is_running():return dict(already_running=True,pid=p.pid)
        except psutil.NoSuchProcess:pass
        except psutil.AccessDenied:return dict(identity_needs_review=True)
        if time.time()-r.get('registered_at',0)<30:return dict(startup_pending=True)
    if (run/'completion_event.json').exists():return dict(terminal_already_present=True)
    args=[sys.executable,str(Path(__file__).resolve()),'watch','--run',str(run)]
    with (folder/'daemon.log').open('ab') as log:
        p=subprocess.Popen(args,cwd=str(ROOT),stdin=subprocess.DEVNULL,stdout=log,stderr=log,creationflags=subprocess.CREATE_NO_WINDOW)
    pp=psutil.Process(p.pid);record=dict(pid=pp.pid,created=pp.create_time(),command=pp.cmdline(),registered_at=time.time());hook.write_json(path,record);return record
def ensure(run):
    run=valid_run(run);folder=run/'lifecycle_observer';folder.mkdir(exist_ok=True)
    with mutex(folder):return _ensure_locked(run)
if __name__=='__main__':
    ap=argparse.ArgumentParser();ap.add_argument('action',choices=['ensure','watch','stop']);ap.add_argument('--run',type=Path,required=True);a=ap.parse_args()
    if a.action=='watch':asyncio.run(watch(a.run))
    elif a.action=='ensure':print(ensure(a.run))
    else:
        folder=valid_run(a.run)/'lifecycle_observer';folder.mkdir(exist_ok=True)
        with mutex(folder):hook.write_json(folder/'stop.json',dict(reason='operator_requested_stop_or_batch_complete',at=time.time()))
        print(dict(observer_stop_requested=True,run=str(a.run)))

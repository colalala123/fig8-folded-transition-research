from pathlib import Path
import json
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import numpy as np
import csv
from response_metrics import analyze,theoretical

ROOT=Path(__file__).resolve().parent
out=ROOT/'stage1';out.mkdir(exist_ok=True)
prior=json.loads((ROOT/'curve_audit.json').read_text(encoding='utf-8'))
records=[]
for r in prior:
    m=analyze(r['path'])
    rec={'case':r['case'],'source_labels':{'uniform_run':r['uniform_run'],'P_legacy':float(r['uniform_P']),
         'R_legacy':float(r['uniform_R']),'uniform_BW_legacy_GHz':float(r['uniform_BW']),
         'note':'Preserved source uniform labels; not folded response labels.'},'metrics':m}
    records.append(rec)
(out/'normalized_history.json').write_text(json.dumps(records,indent=2,ensure_ascii=False),encoding='utf-8')
header='| Case | 源 P | 源 R | 弯折 P_response | 弯折 R_BW | BW_2dB/GHz | BW_joint/GHz | 双峰差/dB | 门槛 |\n|---|---:|---:|---:|---:|---:|---:|---:|---|\n'
for r in records:
    m=r['metrics'];p=m['formal_response_PR'];s=r['source_labels']
    header+=f"| {r['case']} | {s['P_legacy']:.6f} | {s['R_legacy']:.6f} | "+(f"{p['P_response']:.6f} | {p['R_BW']:.6f}" if p else '— | —')+f" | {m['passband_2dB']['BW_GHz']:.6f} | {m['passband_joint']['BW_GHz']:.6f} | {m['peak_difference_dB']:.6f} | {'通过' if p else '未通过'} |\n"
(out/'统一标签表.md').write_text('# 历史曲线标签统一结果\n\n新提取版本 response_pr_v1_20260920；固定 f0_ref=6.85 GHz；R=zm/zs。旧源标签保留，失败样本的正式新标签置空，原始诊断值仍存 JSON。通过表示曲线形状与性能门槛，不代表历史网格收敛已经重新验证。\n\n'+header,encoding='utf-8')
fig,axes=plt.subplots(1,3,figsize=(14,4.4),constrained_layout=True)
for ax,case in zip(axes,['r043','r052','r081']):
    rec=next(r for r in records if r['case']==case);m=rec['metrics'];p=m['formal_response_PR']
    with Path(m['curve_path']).open(encoding='utf-8-sig') as h:rows=list(csv.DictReader(h))
    f=np.array([float(r['Freq_GHz']) for r in rows])
    for key,co in [('S11','#2864a0'),('S21','#bc5a2b')]:
        d=np.array([float(r[key+'_dB']) if key+'_dB' in r else 20*np.log10(float(r[key+'_mag'])) for r in rows])
        ax.plot(f,d,c=co,lw=1.5,label='EM '+key)
    a,b=theoretical(f,p['P_response'],p['R_BW'])
    ax.plot(f,20*np.log10(np.maximum(a,1e-10)),c='#2864a0',ls='--',lw=1,label='Theory S11')
    ax.plot(f,20*np.log10(np.maximum(b,1e-10)),c='#bc5a2b',ls='--',lw=1,label='Theory S21')
    ax.set(xlim=(2.5,11),ylim=(-40,1),xlabel='Frequency (GHz)',ylabel='Magnitude (dB)',title=f"{case}: P={p['P_response']:.3f}, R={p['R_BW']:.3f}")
    ax.grid(alpha=.18);ax.legend(fontsize=8,loc='lower left')
fig.suptitle('Response extraction at fixed f0 = 6.85 GHz; full-curve mismatch remains',fontsize=12)
fig.savefig(out/'response_shape_check.png',dpi=170);plt.close(fig)
print(header)
print('Accepted',sum(r['metrics']['validity']['equal_ripple'] for r in records))

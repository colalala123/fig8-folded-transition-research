def reviewed(run):
    run=Path(run);assert run.resolve().is_relative_to(pc.BASE.resolve())
    p=pc.read(run/'input.json');s=pc.read(run/'state.json');m=pc.read(run/'response_metrics.json')
    a=pc.read(run/'receiver_ack.json');e=hook.load_event(run/'completion_event.json')
    assert e['status']=='completed' and e['target_thread_id']==pc.TARGET
    assert a['event_sha256']==pc.sha(run/'completion_event.json') and a['artifacts_verified'] and a['visual_review_completed'] and a['operation_success']
    assert a['verification_sha256']==pc.sha(run/'receiver_verification.json')
    for x in e['artifacts']:assert pc.sha(run/x['path'])==x['sha256']
    for f,h in p['parent_evidence_hashes'].items():assert pc.sha(f)==h,f
    assert pc.sha(p['source'])==p['source_sha256'] and s['status']=='completed' and s['geometry']==p['geometry']
    assert p['policy_sha256']==pc.sha(pc.HERE/'policy.json') and p['mesh_addendum_sha256']==pc.sha(pc.HERE/'mesh_policy_addendum_v2.json')
    assert 100000<=s['actual_elements']<=112000 and passes(run/'convergence.txt')[-1][1]==s['actual_elements']
    assert s['final_sweep']['InterpTolerance']==1 and not s['final_sweep']['EnforcePassivity']
    assert m['curve_sha256']==pc.sha(run/'sparams.csv') and m['prominent_valley_count']==3
    assert verify_reference(run)==s['reference_treatment']
    return p,m,{str(run/f):pc.sha(run/f) for f in ['input.json','state.json','response_metrics.json','sparams.csv','convergence.txt','receiver_verification.json','receiver_ack.json','completion_event.json','reference_review.json']}

def interior_design(m):return design_verdict(m) and abs(m['central_valley_GHz']-6.72)<=.02+1e-10

def residual_step(j,e):
    j=np.asarray(j,float);e=np.asarray(e,float)
    assert j.shape==(2,2) and e.shape==(2,) and np.isfinite(j).all() and np.isfinite(e).all()
    cond=float(np.linalg.cond(np.diag([1/.5,1/.03])@j@np.diag([.10,.04])))
    assert cond<=100
    d=np.linalg.solve(j,-e)
    scale=min(1.,.10/max(abs(d[0]),1e-30),.04/max(abs(d[1]),1e-30))
    return d*scale,dict(normalized_condition=cond,trust_scale=scale,old_measured_Jacobian=j.tolist(),direct_or_bias_estimated_error=e.tolist(),
        derivative_is_current_center_measured=False,prediction_not_validation=True)

def prepare(case):
    assert case=='D12G05' and not (pc.HERE/'stage_P12_closure.json').exists() and not (pc.HERE/'cases/D12G05/disposition.json').exists()
    assert not dc.solvers()
    roots=['fig8_remediation_A_20261025','fig8_remediation_B_20261025','fig8_remediation_C3_20261025']
    locks=[pc.HERE/'worker.lock',pc.ORIGINAL/'worker.lock',Path('J:/f8_261007/protocol.lock'),Path('J:/f8_261007/pilot_queue.lock')]+[pc.HERE.parent/x/'worker.lock' for x in roots]
    assert not any(p.exists() for p in locks)
    a=pc.read(pc.HERE/'authorization.json');l=pc.read(pc.HERE/'budget_ledger.json');source=pc.read(pc.HERE/'source_review.json')
    assert a['execution_authorized'] and a['stage']=='P12' and a['cases']==ORDER
    assert l['case_cumulative_geometry']==HISTORY and l['new_cached_tasks_reserved']==0 and all(t['status']=='reviewed' for t in l['tasks'])
    n=l['new_geometry_reserved']+1;assert n in (1,2)
    for f,h in source['evidence_hashes'].items():assert pc.sha(f)==h
    parents=dict(source['evidence_hashes']);start=source['geometry'];current=dict(start)
    residual=np.array(source['direct_error_vector'],float)
    previous=None
    if n==2:
        old=pc.BASE/case/job_name(case,1);p,m,h=reviewed(old);parents.update(h)
        assert not interior_design(m),'Select passing interior point for direct, not another trial'
        residual=error(m,6.72)+np.array(source['direct_minus_interpolated_error_bias'],float)
        assert normalized_error(residual)<=.9*normalized_error(np.array(source['direct_error_vector'])),'No justified additional small trial'
        current=p['geometry'];previous=pc.read(pc.HERE/'supervised_release.json')
    delta,model=residual_step(source['borrowed_Jacobian'],residual)
    geometry={**current,'Ls1':current['Ls1']+float(delta[0]),'Lm':current['Lm']+float(delta[1])}
    assert abs(geometry['Ls1']-start['Ls1'])<=.10+1e-12 and abs(geometry['Lm']-start['Lm'])<=.04+1e-12,'Cumulative small step outside disclosed bound'
    assert all(geometry[k]==start[k] for k in ['Wm','Ws','Ls2','Ls3']) and geometry_gate(geometry,source['initial_geometry'])['passed']
    run=pc.BASE/case/job_name(case,n);assert not run.exists()
    prop=dict(action='small_residual_trial',geometry=geometry,delta_from_previous_mm=delta.tolist(),
        cumulative_delta_from_polish_start_mm=[geometry['Ls1']-start['Ls1'],geometry['Lm']-start['Lm']],
        direction_model=model,predicted_remaining_error=(residual+np.array(source['borrowed_Jacobian'])@delta).tolist(),
        note='Old same-policy measured derivatives used only as small direction predictor. Every trial and final direct must actually verify.')
    for f in [pc.HERE/'authorization.json',pc.HERE/'PLAN.md',pc.HERE/'source_review.json',pc.HERE/'reference_validation.json',pc.HERE/'case_reference.py',
              pc.HERE/'geometry_implementation_provenance.json',pc.HERE/'phase_geometry_context.py',pc.HERE/'phase_geometry_publish.py',pc.HERE/'phase_heartbeat.py',
              pc.HERE/'phase_control.py',pc.HERE/'policy.json',pc.HERE/'mesh_policy_addendum_v2.json',pc.HERE/'hook/config.json',Path(__file__).resolve()]:parents[str(f)]=pc.sha(f)
    pc.write(run/'budget_before.json',l);parents[str(run/'budget_before.json')]=pc.sha(run/'budget_before.json')
    pc.write(run/'proposal.json',prop);parents[str(run/'proposal.json')]=pc.sha(run/'proposal.json')
    spec=dict(case=case,role='residual_trial',geometry=geometry,source=source['seed'],source_sha256=source['seed_sha256'],
        run_dir=str(run),input_path=str((run/'input.json').resolve()),project_name=f'D12G05_P12_g{n:02d}',
        policy_sha256=pc.sha(pc.HERE/'policy.json'),mesh_addendum_sha256=pc.sha(pc.HERE/'mesh_policy_addendum_v2.json'),
        new_phase_geometry_ordinal=n,historical_geometry_attempts=11,phase_solve_ordinal_including_history=11+n,
        worker_sha256=pc.sha(pc.HERE/'phase_geometry_worker.py'),controller_sha256=pc.sha(pc.ORIGINAL/'mesh_budget.py'),
        parent_evidence_hashes=parents,created_at=pc.now(),scope='Separately disclosed P12<=2small trials+1direct; old C3 budget/center failure preserved')
    pc.write(run/'input.json',spec)
    if previous:pc.write(pc.HERE/'supervised_releases/first_trial.json',previous)
    pc.write(pc.HERE/'supervised_release.json',dict(case=case,input_path=spec['input_path'],input_sha256=pc.sha(run/'input.json'),bulk_dispatch_released=False,at=pc.now()))
    l['new_geometry_reserved']=n;l['tasks'].append(dict(case=case,role='residual_trial',new_phase_geometry_ordinal=n,lifetime_geometry_ordinal=11+n,path=spec['input_path'],input_sha256=pc.sha(run/'input.json'),status='reserved'))
    pc.write(pc.HERE/'budget_ledger.json',l);print(dict(input=spec['input_path'],geometry=geometry,model=model))

if __name__=='__main__':
    ap=argparse.ArgumentParser();ap.add_argument('--case',choices=ORDER,required=True);prepare(ap.parse_args().case)

class GeometryTests(unittest.TestCase):
    def setUp(self):
        t=tempfile.TemporaryDirectory();self.addCleanup(t.cleanup);p=Path(t.name)/'input.json';p.write_text('{}')
        self.s=dict(case='D12G05',role='residual_trial',new_phase_geometry_ordinal=1,historical_geometry_attempts=11,phase_solve_ordinal_including_history=12,input_path=str(p))
        self.a=dict(execution_authorized=True,stage='P12',cases=ORDER,geometry_cap=2,cached_task_cap=1)
        self.l=dict(case_cumulative_geometry=HISTORY,new_geometry_reserved=1,new_cached_tasks_reserved=0,tasks=[dict(case='D12G05',new_phase_geometry_ordinal=1,path=str(p),input_sha256=sha(p))])
    def test_allowed(self):validate_budget(self.s,self.l,self.a)
    def test_history_not_zero(self):
        self.s['historical_geometry_attempts']=0
        with self.assertRaises(AssertionError):validate_budget(self.s,self.l,self.a)
    def test_only_D12(self):
        self.s['case']='D04G02'
        with self.assertRaises(AssertionError):validate_budget(self.s,self.l,self.a)
    def test_third_trial_forbidden(self):
        self.s['new_phase_geometry_ordinal']=3
        with self.assertRaises(AssertionError):validate_budget(self.s,self.l,self.a)
    def test_old_probe_role_forbidden(self):
        self.s['role']='probe_Ls1'
        with self.assertRaises(AssertionError):validate_budget(self.s,self.l,self.a)
    def test_cache_not_extended(self):
        self.l['new_cached_tasks_reserved']=2
        with self.assertRaises(AssertionError):validate_budget(self.s,self.l,self.a)
    def test_duplicate_forbidden(self):
        self.l['tasks'].append(copy.deepcopy(self.l['tasks'][0]))
        with self.assertRaises(AssertionError):validate_budget(self.s,self.l,self.a)
    def test_recovery_not_implicit(self):
        self.s['resume_checkpoint']='anything'
        with self.assertRaises(AssertionError):validate_budget(self.s,self.l,self.a)

class StepTests(unittest.TestCase):
    def test_actual_direct_residual_prediction(self):
        v=pc.read(pc.HERE/'source_review.json');d,m=residual_step(v['borrowed_Jacobian'],v['direct_error_vector'])
        self.assertTrue(np.allclose(d,[.06178186380756949,.020990084718525783]))
        self.assertTrue(np.allclose(np.array(v['borrowed_Jacobian'])@d+v['direct_error_vector'],[0,0],atol=1e-12))
        self.assertFalse(m['derivative_is_current_center_measured'])
    def test_huge_step_whole_scaled(self):
        d,m=residual_step([[10,-20],[.1,-2]],[-100,1])
        self.assertLessEqual(abs(d[0]),.10+1e-12);self.assertLessEqual(abs(d[1]),.04+1e-12)
    def test_singular_direction_rejected(self):
        with self.assertRaises(AssertionError):residual_step([[1,1],[2,2]],[1,1])
    def test_nonfinite_rejected(self):
        with self.assertRaises(AssertionError):residual_step([[1,0],[0,1]],[float('nan'),1])
    def test_boundary_not_selected_for_polish_direct(self):
        with patch('prepare_geometry.design_verdict',return_value=True):
            self.assertFalse(interior_design(dict(central_valley_GHz=6.75)))
            self.assertTrue(interior_design(dict(central_valley_GHz=6.72)))

if __name__=='__main__':unittest.main()

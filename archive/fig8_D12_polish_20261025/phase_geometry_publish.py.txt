async def publish(run,case,status,summary,artifacts):
    cfg=hook.load_config(HERE/'hook/config.json');assert cfg['default_thread_id']==TARGET and cfg['authorized_threads']==[TARGET] and case=='D12G05'
    path=run/'completion_event.json'
    if not path.exists():path=hook.create_event(run,status,summary,artifacts,config=cfg,verified_success=status=='completed',purpose='Existing human all3goal, disclosed finiteD12residual polish only')
    hook.prompt_for=prompt;receipt=await hook.deliver(path,config=cfg)
    if receipt.get('state')=='accepted':
        from phase_heartbeat import arm
        try:arm(path)
        except Exception as exc:write(run/'heartbeat_start_error.json',dict(at=now(),error=repr(exc)))
    return receipt

def interpolate(rows,wm,ws,arm):
    xs=sorted({r['geometry_mm']['Wm'] for r in rows});ys=sorted({r['geometry_mm']['Ws'] for r in rows})
    x0=max(x for x in xs if x<=wm);x1=min(x for x in xs if x>=wm)
    y0=max(y for y in ys if y<=ws);y1=min(y for y in ys if y>=ws)
    tx=(wm-x0)/(x1-x0);ty=(ws-y0)/(y1-y0)
    terms=[]
    for x,wx in [(x0,1-tx),(x1,tx)]:
        for y,wy in [(y0,1-ty),(y1,ty)]:
            r=next(r for r in rows if r['geometry_mm']['Wm']==x and r['geometry_mm']['Ws']==y and abs(r['geometry_mm']['Ls3']-y-arm)<1e-9)
            terms.append(dict(case=r['case'],weight=wx*wy,Lm=r['geometry_mm']['Lm'],Ls1=r['geometry_mm']['Ls1'],result_sha256=r['result_sha256']))
    assert abs(sum(t['weight'] for t in terms)-1)<1e-12
    return {k:sum(t['weight']*t[k] for t in terms) for k in ['Lm','Ls1']},terms
